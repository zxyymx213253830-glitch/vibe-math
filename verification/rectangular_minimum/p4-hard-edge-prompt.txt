private def p1 (n : ℕ) (x : ℝ) : ℝ :=
  ∑ l ∈ Finset.range (n+1), (n.choose l : ℝ) / (l.factorial : ℝ) * l * x^(l-1)
private def p2 (n : ℕ) (x : ℝ) : ℝ :=
  ∑ l ∈ Finset.range (n+1), (n.choose l : ℝ) / (l.factorial : ℝ) * l * (l-1 : ℕ) * x^(l-2)

private theorem dP (n : ℕ) (x : ℝ) :
    HasDerivAt (oneExtraRowMinimumPolynomial n) (p1 n x) x := by
  unfold oneExtraRowMinimumPolynomial p1
  convert HasDerivAt.fun_sum (fun l (_ : l ∈ Finset.range (n+1)) =>
    ((hasDerivAt_pow l x).const_mul ((n.choose l : ℝ) / (l.factorial : ℝ)))) using 1 <;>
    simp only [mul_div_assoc, div_mul_eq_mul_div, mul_assoc]

private theorem dP1 (n : ℕ) (x : ℝ) : HasDerivAt (p1 n) (p2 n x) x := by
  unfold p1 p2
  convert HasDerivAt.fun_sum (fun l (_ : l ∈ Finset.range (n+1)) =>
    ((hasDerivAt_pow (l-1) x).const_mul ((n.choose l : ℝ) / (l.factorial : ℝ) * l))) using 1 <;>
    simp only [Nat.sub_sub, one_add_one_eq_two, mul_assoc]

private theorem P0 (n : ℕ) : oneExtraRowMinimumPolynomial n 0 = 1 := by
  unfold oneExtraRowMinimumPolynomial
  rw [Finset.sum_eq_single 0]
  · simp
  · intro l hl hne
    simp [zero_pow hne]
  · simp

private theorem P10 (n : ℕ) (hn : 0 < n) : p1 n 0 = n := by
  unfold p1
  rw [Finset.sum_eq_single 1]
  · simp
  · intro l hl hne
    by_cases h0 : l = 0
    · simp [h0]
    · have hp : l-1 ≠ 0 := by omega
      simp [zero_pow hp]
  · intro h
    have : 1 ∈ Finset.range (n+1) := Finset.mem_range.mpr (by omega)
    exact (h this).elim

private theorem P20 (n : ℕ) : p2 n 0 = (n.choose 2 : ℝ) := by
  unfold p2
  rw [Finset.sum_eq_single 2]
  · norm_num
  · intro l hl hne
    by_cases h0 : l = 0
    · simp [h0]
    by_cases h1 : l = 1
    · simp [h1]
    have hp : l-2 ≠ 0 := by omega
    simp [zero_pow hp]
  · intro h
    have hn : n < 2 := by simpa using h
    simp [Nat.choose_eq_zero_of_lt hn]

def oneExtraRowMinimumPolynomial (n : ℕ) (x : ℝ) : ℝ :=
  ∑ l ∈ Finset.range (n + 1), (n.choose l : ℝ) * x ^ l / ((Nat.factorial l : ℕ) : ℝ)

theorem sum_finsets_card_function (n : ℕ) (f : ℕ → ℝ) :
    (∑ S : Finset (Fin n), f S.card) =
      ∑ l ∈ Finset.range (n + 1), (n.choose l : ℝ) * f l := by
  classical
  rw [← Finset.powerset_univ, Finset.sum_powerset]
  simp_rw [Finset.sum_powersetCard]
  simp only [Finset.card_univ, Fintype.card_fin, nsmul_eq_mul]

theorem sum_finsets_complement_card_function (n : ℕ) (f : ℕ → ℝ) :
    (∑ S : Finset (Fin n), f (n - S.card)) =
      ∑ l ∈ Finset.range (n + 1), (n.choose l : ℝ) * f l := by
  classical
  let e : Finset (Fin n) ≃ Finset (Fin n) :=
    ⟨fun S => Sᶜ, fun S => Sᶜ, fun S => compl_compl S, fun S => compl_compl S⟩
  have h := Equiv.sum_comp e (fun S : Finset (Fin n) => f S.card)
  simpa only [e, Equiv.coe_fn_mk, Finset.card_compl, Fintype.card_fin,
    sum_finsets_card_function] using h

theorem oneExtraRowMinimumPolynomial_eq_subsets (n : ℕ) (x : ℝ) :
    oneExtraRowMinimumPolynomial n x =
      ∑ S : Finset (Fin n), x ^ (n - S.card) /
        ((Nat.factorial (n - S.card) : ℕ) : ℝ) := by
  have h := sum_finsets_complement_card_function n
    (fun l => x ^ l / ((Nat.factorial l : ℕ) : ℝ))
  simpa only [oneExtraRowMinimumPolynomial, mul_div_assoc] using h.symm
