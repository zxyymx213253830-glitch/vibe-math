def oneExtraRowMinimumPolynomial (n : ℕ) (x : ℝ) : ℝ :=
  ∑ l ∈ Finset.range (n + 1), (n.choose l : ℝ) * x ^ l / ((Nat.factorial l : ℕ) : ℝ)

theorem sum_finsets_card_function (n : ℕ) (f : ℕ → ℝ) :
    (∑ S : Finset (Fin n), f S.card) =
      ∑ l ∈ Finset.range (n + 1), (n.choose l : ℝ) * f l := by
  classical
  rw [← Finset.powerset_univ, Finset.sum_powerset]
  simp_rw [Finset.sum_powersetCard]
  simp only [Finset.card_univ, Fintype.card_fin, nsmul_eq_mul]

theorem sum_finsets_complement_card_function (n : ℕ) (f : ℕ → ℝ) :
    (∑ S : Finset (Fin n), f (n - S.card)) =
      ∑ l ∈ Finset.range (n + 1), (n.choose l : ℝ) * f l := by
  classical
  let e : Finset (Fin n) ≃ Finset (Fin n) :=
    ⟨fun S => Sᶜ, fun S => Sᶜ, fun S => compl_compl S, fun S => compl_compl S⟩
  have h := Equiv.sum_comp e (fun S : Finset (Fin n) => f S.card)
  simpa only [e, Equiv.coe_fn_mk, Finset.card_compl, Fintype.card_fin,
    sum_finsets_card_function] using h

theorem oneExtraRowMinimumPolynomial_eq_subsets (n : ℕ) (x : ℝ) :
    oneExtraRowMinimumPolynomial n x =
      ∑ S : Finset (Fin n), x ^ (n - S.card) /
        ((Nat.factorial (n - S.card) : ℕ) : ℝ) := by
  have h := sum_finsets_complement_card_function n
    (fun l => x ^ l / ((Nat.factorial l : ℕ) : ℝ))
  simpa only [oneExtraRowMinimumPolynomial, mul_div_assoc] using h.symm

theorem oneExtraRowMinimumPolynomial_nonneg (n : ℕ) (x : ℝ) (hx : 0 ≤ x) :
    0 ≤ oneExtraRowMinimumPolynomial n x := by
  unfold oneExtraRowMinimumPolynomial
  exact Finset.sum_nonneg (fun _ _ => by positivity)
