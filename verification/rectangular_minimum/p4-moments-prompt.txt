theorem gammaPDF_nat_power_shift (k p : ℕ) (r : ℝ) (hr : 0 < r) :
    gammaPDF ((k + 1 : ℕ) : ℝ) 1 r * ENNReal.ofReal (r ^ p) =
      ENNReal.ofReal (((Nat.factorial (k + p) : ℕ) : ℝ) /
        ((Nat.factorial k : ℕ) : ℝ)) * gammaPDF ((k + p + 1 : ℕ) : ℝ) 1 r := by
  rw [gammaPDF_nat_succ_positive k r hr, gammaPDF_nat_succ_positive (k + p) r hr,
    ← ENNReal.ofReal_mul (by positivity), ← ENNReal.ofReal_mul (by positivity)]
  congr 1
  rw [pow_add]
  have hk : ((Nat.factorial k : ℕ) : ℝ) ≠ 0 := by positivity
  have hkp : ((Nat.factorial (k + p) : ℕ) : ℝ) ≠ 0 := by positivity
  field_simp

theorem gammaMeasure_nat_power_bias (k p : ℕ) :
    (gammaMeasure ((k + 1 : ℕ) : ℝ) 1).withDensity (fun r => ENNReal.ofReal (r ^ p)) =
      ENNReal.ofReal (((Nat.factorial (k + p) : ℕ) : ℝ) /
        ((Nat.factorial k : ℕ) : ℝ)) • gammaMeasure ((k + p + 1 : ℕ) : ℝ) 1 := by
  have hG : Measurable (gammaPDF ((k + 1 : ℕ) : ℝ) 1) :=
    ENNReal.measurable_ofReal.comp (measurable_gammaPDFReal _ _)
  have hP : Measurable (fun r : ℝ => ENNReal.ofReal (r ^ p)) := by fun_prop
  rw [gammaMeasure_nat_succ_eq_positive_density k,
    gammaMeasure_nat_succ_eq_positive_density (k + p),
    ← withDensity_mul _ hG hP, ← withDensity_smul' _ _ ENNReal.ofReal_ne_top]
  apply withDensity_congr_ae
  filter_upwards [ae_restrict_mem measurableSet_Ioi] with r hr
  exact gammaPDF_nat_power_shift k p r hr

theorem lintegral_gammaMeasure_nat_power (k p : ℕ) :
    ∫⁻ r, ENNReal.ofReal (r ^ p) ∂gammaMeasure ((k + 1 : ℕ) : ℝ) 1 =
      ENNReal.ofReal (((Nat.factorial (k + p) : ℕ) : ℝ) /
        ((Nat.factorial k : ℕ) : ℝ)) := by
  letI : IsProbabilityMeasure (gammaMeasure ((k + p + 1 : ℕ) : ℝ) 1) :=
    isProbabilityMeasure_gammaMeasure (by positivity) (by norm_num)
  have h := congrArg (fun μ : Measure ℝ => μ Set.univ) (gammaMeasure_nat_power_bias k p)
  rw [withDensity_apply _ MeasurableSet.univ, Measure.restrict_univ,
    Measure.smul_apply, measure_univ] at h
  simpa only [smul_eq_mul, mul_one] using h

theorem lintegral_gammaMeasure_nat_abs_power (k p : ℕ) :
    ∫⁻ r, ENNReal.ofReal (|r| ^ p) ∂gammaMeasure ((k + 1 : ℕ) : ℝ) 1 =
      ENNReal.ofReal (((Nat.factorial (k + p) : ℕ) : ℝ) /
        ((Nat.factorial k : ℕ) : ℝ)) := by
  rw [← lintegral_gammaMeasure_nat_power k p]
  apply lintegral_congr_ae
  have hp : ∀ᵐ r ∂gammaMeasure ((k + 1 : ℕ) : ℝ) 1, 0 < r :=
    (gammaMeasure_nat_succ_absolutelyContinuous_positive k).ae_le
      (ae_restrict_mem measurableSet_Ioi)
  filter_upwards [hp] with r hr
  rw [abs_of_pos hr]

end

theorem pi_gaussianReal_map_injective {ι κ : Type*} [Fintype ι] [Fintype κ]
    (e : κ → ι) (he : Function.Injective e) :
    (Measure.pi (fun _ : ι => gaussianReal 0 1)).map (fun x => fun i => x (e i)) =
      Measure.pi (fun _ : κ => gaussianReal 0 1) := by
  classical
  letI : Fintype (Set.range e) := Subtype.fintype (fun i => i ∈ Set.range e)
  let eR : κ ≃ Set.range e := Equiv.ofInjective e he
  have hs := measurePreserving_piEquivPiSubtypeProd
    (fun _ : ι => gaussianReal 0 1) (fun i => i ∈ Set.range e)
  have hp := (measurePreserving_fst
    (μ := Measure.pi (fun _ : Set.range e => gaussianReal 0 1))
    (ν := Measure.pi (fun _ : {i : ι // i ∉ Set.range e} => gaussianReal 0 1))).comp hs
  have hr := measurePreserving_piCongrLeft (fun _ : Set.range e => gaussianReal 0 1) eR
  exact (hr.symm.comp hp).map_eq

theorem stdGaussian_map_injective_coordinates {ι κ : Type*} [Fintype ι] [Fintype κ]
    (e : κ → ι) (he : Function.Injective e) :
    (stdGaussian (EuclideanSpace ℝ ι)).map
      (fun z => (WithLp.toLp 2 (fun i => z (e i)) : EuclideanSpace ℝ κ)) =
        stdGaussian (EuclideanSpace ℝ κ) := by
  rw [← map_pi_eq_stdGaussian, Measure.map_map (by fun_prop) (by fun_prop)]
  have hf : (fun z : EuclideanSpace ℝ ι =>
      (WithLp.toLp 2 (fun i => z (e i)) : EuclideanSpace ℝ κ)) ∘ WithLp.toLp 2 =
      WithLp.toLp 2 ∘ (fun z : ι → ℝ => fun i => z (e i)) := rfl
  rw [hf, ← Measure.map_map (by fun_prop) (by fun_prop),
    pi_gaussianReal_map_injective e he, map_pi_eq_stdGaussian]

def sampleColumnSelection {m n k : ℕ} (e : Fin k → Fin n)
    (z : ComplexSample (m := m) (n := n)) : ComplexSample (m := m) (n := k) :=
  WithLp.toLp 2 (fun ijl => z ((ijl.1.1, e ijl.1.2), ijl.2))

@[fun_prop]
theorem measurable_sampleColumnSelection {m n k : ℕ} (e : Fin k → Fin n) :
    Measurable (sampleColumnSelection (m := m) e) := by
  unfold sampleColumnSelection
  fun_prop

theorem stdGaussian_sampleColumnSelection_law {m n k : ℕ}
    (e : Fin k → Fin n) (he : Function.Injective e) :
    (stdGaussian (ComplexSample (m := m) (n := n))).map (sampleColumnSelection e) =
      stdGaussian (ComplexSample (m := m) (n := k)) := by
  apply stdGaussian_map_injective_coordinates
  rintro ⟨⟨i, j⟩, l⟩ ⟨⟨i', j'⟩, l'⟩ h
  simp only [Prod.mk.injEq] at h
  rcases h with ⟨⟨hi, hj⟩, hl⟩
  simp only [hi, he hj, hl]

theorem lintegral_centralBartlettGammaProduct_norm_power (d n p : ℕ) :
    ∫⁻ z, ENNReal.ofReal (‖z‖ ^ p) ∂centralBartlettGammaProductLaw d n =
      ∏ j : Fin n, ENNReal.ofReal
        (((Nat.factorial (d + j.val + p) : ℕ) : ℝ) / ((Nat.factorial (d + j.val) : ℕ) : ℝ)) := by
  induction n with
  | zero => simp [centralBartlettGammaProductLaw_zero]
  | succ n ih =>
      rw [centralBartlettGammaProductLaw_succ, lintegral_map (by fun_prop) (by fun_prop)]
      have hf : (fun q : ℝ × ℂ => ENNReal.ofReal (‖(q.1 : ℂ) * q.2‖ ^ p)) =
          (fun q => ENNReal.ofReal (|q.1| ^ p) * ENNReal.ofReal (‖q.2‖ ^ p)) := by
        funext q
        rw [norm_mul, mul_pow, Complex.norm_real, Real.norm_eq_abs,
          ENNReal.ofReal_mul (by positivity)]
      change ∫⁻ q : ℝ × ℂ, ENNReal.ofReal (‖(q.1 : ℂ) * q.2‖ ^ p)
        ∂(gammaMeasure ((d + n + 1 : ℕ) : ℝ) 1).prod (centralBartlettGammaProductLaw d n) = _
      rw [hf, lintegral_prod_mul (f := fun r : ℝ => ENNReal.ofReal (|r| ^ p))
        (g := fun z : ℂ => ENNReal.ofReal (‖z‖ ^ p)) (by fun_prop) (by fun_prop),
        lintegral_gammaMeasure_nat_abs_power, ih, Fin.prod_univ_castSucc]
      simp only [Fin.val_castSucc, Fin.val_last]
      rw [mul_comm]

theorem complexWishart_determinant_norm_moment (d n p : ℕ) :
    ∫⁻ W, ENNReal.ofReal (‖W.det‖ ^ p) ∂complexWishart (m := d + n) (n := n) =
      ∏ j : Fin n, ENNReal.ofReal
        (((Nat.factorial (d + j.val + p) : ℕ) : ℝ) / ((Nat.factorial (d + j.val) : ℕ) : ℝ)) := by
  rw [← lintegral_centralBartlettGammaProduct_norm_power d n p,
    ← complexWishart_det_eq_gammaProduct, lintegral_map (by fun_prop) (by fun_prop)]

theorem factorial_ratio_product (d n : ℕ) :
    (∏ j : Fin n, (((Nat.factorial (d + j.val + 1) : ℕ) : ℝ) /
      ((Nat.factorial (d + j.val) : ℕ) : ℝ))) =
      ((Nat.factorial (d + n) : ℕ) : ℝ) / ((Nat.factorial d : ℕ) : ℝ) := by
  induction n with
  | zero => simp [show ((Nat.factorial d : ℕ) : ℝ) ≠ 0 by positivity]
  | succ n ih =>
      rw [Fin.prod_univ_castSucc]
      simp only [Fin.val_castSucc, Fin.val_last]
      rw [ih]
      have hd : ((Nat.factorial d : ℕ) : ℝ) ≠ 0 := by positivity
      have hdn : ((Nat.factorial (d + n) : ℕ) : ℝ) ≠ 0 := by positivity
      field_simp
      simp only [Nat.add_assoc]

theorem complexWishart_determinant_norm_mean (d n : ℕ) :
    ∫⁻ W, ENNReal.ofReal ‖W.det‖ ∂complexWishart (m := d + n) (n := n) =
      ENNReal.ofReal (((Nat.factorial (d + n) : ℕ) : ℝ) /
        ((Nat.factorial d : ℕ) : ℝ)) := by
  have h := complexWishart_determinant_norm_moment d n 1
  simp only [pow_one] at h
  rw [h, ← ENNReal.ofReal_prod_of_nonneg (fun j _ => by positivity), factorial_ratio_product]

theorem complexWishart_determinant_norm_mean_of_le (t s : ℕ) (hst : s ≤ t) :
    ∫⁻ W, ENNReal.ofReal ‖W.det‖ ∂complexWishart (m := t) (n := s) =
      ENNReal.ofReal (((Nat.factorial t : ℕ) : ℝ) /
        ((Nat.factorial (t - s) : ℕ) : ℝ)) := by
  simpa only [Nat.sub_add_cancel hst] using complexWishart_determinant_norm_mean (t - s) s

end
