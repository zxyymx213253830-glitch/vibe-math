variable {α : Type*} [MeasurableSpace α] (μ : Measure α) [IsProbabilityMeasure μ]

def probabilityTail (f : α → ℝ) (t : ℝ) : ℝ := μ.real {a | t ≤ f a}

theorem probabilityTail_nonneg (f : α → ℝ) (t : ℝ) : 0 ≤ probabilityTail μ f t :=
  ENNReal.toReal_nonneg

theorem probabilityTail_le_one (f : α → ℝ) (t : ℝ) : probabilityTail μ f t ≤ 1 := by
  exact measureReal_le_one

theorem probabilityTail_antitone (f : α → ℝ) : Antitone (probabilityTail μ f) := by
  intro s t hst
  apply ENNReal.toReal_mono (measure_ne_top _ _)
  apply measure_mono
  exact fun a ha => hst.trans ha

theorem measurable_probabilityTail (f : α → ℝ) : Measurable (probabilityTail μ f) :=
  (probabilityTail_antitone μ f).measurable

theorem integrable_nonneg_iff_probabilityTail (f : α → ℝ) (hf : Measurable f)
    (hnn : 0 ≤ᵐ[μ] f) :
    Integrable f μ ↔ IntegrableOn (probabilityTail μ f) (Ioi 0) := by
  have htail : (∫⁻ t in Ioi (0 : ℝ), ENNReal.ofReal (probabilityTail μ f t)) =
      ∫⁻ a, ENNReal.ofReal (f a) ∂μ := by
    rw [lintegral_eq_lintegral_meas_le μ hnn hf.aemeasurable]
    apply lintegral_congr_ae
    filter_upwards [] with t
    exact ENNReal.ofReal_toReal (measure_ne_top _ _)
  change Integrable f μ ↔ Integrable (probabilityTail μ f) (volume.restrict (Ioi 0))
  rw [← lintegral_ofReal_ne_top_iff_integrable hf.aestronglyMeasurable hnn,
    ← lintegral_ofReal_ne_top_iff_integrable
      (measurable_probabilityTail μ f).aestronglyMeasurable
      (ae_of_all _ (probabilityTail_nonneg μ f)),htail]

end
end JinWishart

theorem integrableOn_positive_sandwich_iff (F g : ℝ → ℝ) (s : Set ℝ)
    (hF : Measurable F) (hg : Measurable g)
    (hFn : ∀ t ∈ s, 0 ≤ F t) (hgn : ∀ t ∈ s, 0 ≤ g t)
    (c C : ℝ) (hc : 0 < c) (hC : 0 < C)
    (hb : ∀ t ∈ s, c*g t ≤ F t ∧ F t ≤ C*g t) (hs : MeasurableSet s) :
    IntegrableOn F s ↔ IntegrableOn g s := by
  constructor
  · intro h
    apply (h.const_mul c⁻¹).mono' hg.aestronglyMeasurable
    filter_upwards [ae_restrict_mem hs] with t ht
    rw [Real.norm_eq_abs,abs_of_nonneg (hgn t ht)]
    rw [← div_eq_inv_mul]
    apply (le_div_iff₀ hc).mpr
    simpa only [mul_comm] using (hb t ht).1
  · intro h
    apply (h.const_mul C).mono' hF.aestronglyMeasurable
    filter_upwards [ae_restrict_mem hs] with t ht
    rw [Real.norm_eq_abs,abs_of_nonneg (hFn t ht)]
    exact (hb t ht).2

theorem integrableOn_natPowerRatio_iff (q k : ℕ) :
    IntegrableOn (fun t : ℝ => t^q/t^k) (Ioi 4) ↔ q+1<k := by
  have he : EqOn (fun t : ℝ => t^q/t^k) (fun t : ℝ => t^((q : ℝ)-(k : ℝ))) (Ioi 4) := by
    intro t ht
    have ht4 : (4 : ℝ)<t := ht
    dsimp only
    rw [Real.rpow_sub (by linarith : 0 < t),Real.rpow_natCast,Real.rpow_natCast]
  rw [integrableOn_congr_fun he measurableSet_Ioi,integrableOn_Ioi_rpow_iff (by norm_num : (0 : ℝ)<4)]
  constructor
  · intro h
    exact_mod_cast (by linarith : (q : ℝ)+1<(k : ℝ))
  · intro h
    have hr : (q : ℝ)+1<(k : ℝ) := by exact_mod_cast h
    linarith

theorem integrableOn_powerTail_iff (F : ℝ → ℝ) (hF : Measurable F)
    (hFn : ∀ t ∈ Ioi (4 : ℝ), 0 ≤ F t) (q k : ℕ)
    (c C : ℝ) (hc : 0 < c) (hC : 0 < C)
    (hb : ∀ t ∈ Ioi (4 : ℝ), c*(t^q/t^k) ≤ F t ∧ F t ≤ C*(t^q/t^k)) :
    IntegrableOn F (Ioi 4) ↔ q+1<k := by
  rw [integrableOn_positive_sandwich_iff F (fun t : ℝ => t^q/t^k) (Ioi 4)
    hF (by fun_prop) hFn (by
      intro t ht
      have ht4 : (4 : ℝ)<t := ht
      have ht0 : 0<t := by linarith
      exact div_nonneg (pow_nonneg ht0.le q) (pow_nonneg ht0.le k))
    c C hc hC hb measurableSet_Ioi]
  exact integrableOn_natPowerRatio_iff q k

end
end JinWishart

open Set MeasureTheory

noncomputable section
variable {α : Type*} [MeasurableSpace α] (μ : Measure α) [IsProbabilityMeasure μ]

theorem probabilityTail_integrableOn_near_zero (f : α → ℝ) :
    IntegrableOn (probabilityTail μ f) (Ioc (0 : ℝ) 4) := by
  apply (integrable_const (1 : ℝ)).mono' (measurable_probabilityTail μ f).aestronglyMeasurable
  filter_upwards [] with t
  rw [Real.norm_eq_abs,abs_of_nonneg (probabilityTail_nonneg μ f t)]
  exact probabilityTail_le_one μ f t

theorem integrable_nonneg_iff_probabilityTail_Ioi_four (f : α → ℝ)
    (hf : Measurable f) (hnn : 0 ≤ᵐ[μ] f) :
    Integrable f μ ↔ IntegrableOn (probabilityTail μ f) (Ioi 4) := by
  rw [integrable_nonneg_iff_probabilityTail μ f hf hnn,
    ← Ioc_union_Ioi_eq_Ioi (by norm_num : (0 : ℝ)≤4),integrableOn_union]
  simp only [probabilityTail_integrableOn_near_zero,true_and]

theorem integrable_nonneg_iff_two_sided_power_tail (f : α → ℝ)
    (hf : Measurable f) (hnn : 0 ≤ᵐ[μ] f) (k : ℕ)
    (c C : ℝ) (hc : 0 < c) (hC : 0 < C)
    (hb : ∀ t : ℝ, 4 ≤ t → c/t^k ≤ probabilityTail μ f t ∧ probabilityTail μ f t ≤ C/t^k) :
    Integrable f μ ↔ 1<k := by
  rw [integrable_nonneg_iff_probabilityTail_Ioi_four μ f hf hnn]
  have h := integrableOn_powerTail_iff (probabilityTail μ f) (measurable_probabilityTail μ f)
    (fun t _ => probabilityTail_nonneg μ f t) 0 k c C hc hC (by
      intro t ht
      have ht4 : (4 : ℝ)<t := ht
      simpa only [pow_zero,div_eq_mul_inv,one_mul] using hb t ht4.le)
  simpa only [zero_add] using h

end
end JinWishart
