theorem norm_sub_square_le_twice {E : Type*} [NormedAddCommGroup E] (x y : E) :
    ‖x-y‖^2 ≤ 2*‖x‖^2+2*‖y‖^2 := by
  have h := pow_le_pow_left₀ (norm_nonneg (x-y)) (norm_sub_le x y) 2
  nlinarith [sq_nonneg (‖x‖-‖y‖)]

theorem gaussian_shift_exponent_bounds {E : Type*} [NormedAddCommGroup E] (x y : E) :
    -‖y‖^2-2*‖x‖^2 ≤ -‖x-y‖^2/2 ∧
    -‖x-y‖^2/2 ≤ ‖y‖^2-‖x‖^2/8 := by
  have h1 := norm_sub_square_le_twice x y
  have h2 := norm_sub_square_le_twice (x-y) (-y)
  simp only [sub_neg_eq_add,sub_add_cancel,norm_neg] at h2
  constructor <;> nlinarith [sq_nonneg ‖x‖,sq_nonneg ‖y‖]

theorem euclideanStdGaussianDensity_shift_bounds {ι : Type*} [Fintype ι]
    (x y : EuclideanSpace ℝ ι) :
    Real.exp (-‖y‖^2)*euclideanStdGaussianDensity ((2 : ℝ) • x) ≤
        euclideanStdGaussianDensity (x-y) ∧
    euclideanStdGaussianDensity (x-y) ≤
        Real.exp (‖y‖^2)*euclideanStdGaussianDensity ((1/2 : ℝ) • x) := by
  have h := gaussian_shift_exponent_bounds x y
  have hnorm2 : ‖(2 : ℝ) • x‖^2 = 4*‖x‖^2 := by
    rw [norm_smul,Real.norm_eq_abs]; norm_num; ring
  have hnormhalf : ‖(1/2 : ℝ) • x‖^2 = ‖x‖^2/4 := by
    rw [norm_smul,Real.norm_eq_abs]; norm_num; ring
  have hc : 0 ≤ (Real.sqrt (2*Real.pi))⁻¹ ^ Fintype.card ι := by positivity
  unfold euclideanStdGaussianDensity
  rw [hnorm2,hnormhalf]
  constructor
  · have he : Real.exp (-‖y‖^2)*Real.exp (-(4*‖x‖^2)/2) ≤
        Real.exp (-‖x-y‖^2/2) := by
      rw [← Real.exp_add]
      apply Real.exp_le_exp.mpr
      nlinarith [h.1]
    simpa only [mul_assoc,mul_left_comm,mul_comm] using mul_le_mul_of_nonneg_left he hc
  · have he : Real.exp (-‖x-y‖^2/2) ≤
        Real.exp (‖y‖^2)*Real.exp (-(‖x‖^2/4)/2) := by
      rw [← Real.exp_add]
      apply Real.exp_le_exp.mpr
      nlinarith [h.2]
    simpa only [mul_assoc,mul_left_comm,mul_comm] using mul_le_mul_of_nonneg_left he hc

end

theorem stdGaussian_smul_eq_density {ι : Type*} [Fintype ι]
    (r : ℝ) (hr : r ≠ 0) :

      volume.withDensity (fun z => ENNReal.ofReal (abs (r^Fintype.card ι)⁻¹)*
        ENNReal.ofReal (euclideanStdGaussianDensity (r⁻¹ • z))) := by
  simpa only [add_zero,sub_zero] using
    stdGaussian_scaled_shift_eq_withDensity r hr (0 : EuclideanSpace ℝ ι)

theorem stdGaussian_shift_measure_lower {ι : Type*} [Fintype ι]
    (y : EuclideanSpace ℝ ι) :
    ENNReal.ofReal (((2 : ℝ)^Fintype.card ι)⁻¹*Real.exp (-‖y‖^2)) •
      (stdGaussian (EuclideanSpace ℝ ι)).map (fun z => (1/2 : ℝ) • z) ≤
    (stdGaussian (EuclideanSpace ℝ ι)).map (fun z => z+y) := by
  rw [stdGaussian_smul_eq_density (1/2) (by norm_num),stdGaussian_shift_eq_withDensity]
  rw [← withDensity_smul _ (by unfold euclideanStdGaussianDensity; fun_prop)]
  apply withDensity_mono
  apply ae_of_all
  intro z
  have hj : abs (((1/2 : ℝ)^Fintype.card ι)⁻¹) = (2 : ℝ)^Fintype.card ι := by
    simp only [one_div,inv_pow,inv_inv,abs_of_nonneg (by positivity : (0 : ℝ) ≤ 2^Fintype.card ι)]
  dsimp only [Pi.smul_apply,smul_eq_mul]
  rw [hj]
  norm_num only [one_div,inv_inv]
  rw [← ENNReal.ofReal_mul (by positivity),← ENNReal.ofReal_mul (by positivity)]
  apply ENNReal.ofReal_le_ofReal
  convert (euclideanStdGaussianDensity_shift_bounds z y).1 using 1
  field_simp
