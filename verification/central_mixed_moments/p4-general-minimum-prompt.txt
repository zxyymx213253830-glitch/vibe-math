abbrev ComplexMatrixPolynomial (m n : ℕ) :=
  MvPolynomial ((Fin m × Fin n) ⊕ (Fin m × Fin n)) ℂ

def gramEntryPolynomial (m n : ℕ) : Matrix (Fin n) (Fin n) (ComplexMatrixPolynomial m n) :=
  fun i j => ∑ k, MvPolynomial.X (Sum.inr (k,i)) * MvPolynomial.X (Sum.inl (k,j))

def shiftedGramDeterminantPolynomial (m n d : ℕ) (x : ℝ) : ComplexMatrixPolynomial m n :=
  (gramEntryPolynomial m n + (MvPolynomial.C (x : ℂ) : ComplexMatrixPolynomial m n) • 1).det ^ d

theorem eval_gramEntryPolynomial (m n : ℕ)
    (z : ComplexSample (m := m) (n := n)) (i j : Fin n) :
    complexGaussianPolynomialEval (gramEntryPolynomial m n i j) (complexMatrixEntries z) =
      complexSampleGram z i j := by
  classical
  simp [complexGaussianPolynomialEval,gramEntryPolynomial,complexMatrixEntries,
    complexSampleGram,complexGram,Matrix.mul_apply,Matrix.conjTranspose_apply]

theorem eval_shiftedGramDeterminantPolynomial (m n d : ℕ) (x : ℝ)
    (z : ComplexSample (m := m) (n := n)) :
    complexGaussianPolynomialEval (shiftedGramDeterminantPolynomial m n d x)
      (complexMatrixEntries z) = (complexSampleGram z+(x : ℂ) • 1).det ^ d := by
  classical
  let e := MvPolynomial.eval (Sum.elim (complexMatrixEntries z)
    (fun i => star (complexMatrixEntries z i)))
  change e ((gramEntryPolynomial m n+(MvPolynomial.C (x : ℂ) : ComplexMatrixPolynomial m n) • 1).det ^ d) = _
  rw [map_pow,e.map_det]
  congr 2
  ext i j
  change e (gramEntryPolynomial m n i j+(((MvPolynomial.C (x : ℂ) : ComplexMatrixPolynomial m n)) •
    (1 : Matrix (Fin n) (Fin n) (ComplexMatrixPolynomial m n))) i j) = _
  rw [map_add]
  have h := eval_gramEntryPolynomial m n z i j
  change e (gramEntryPolynomial m n i j) = _ at h
  rw [h]
  simp [e,Matrix.smul_apply,Matrix.one_apply]
  split_ifs <;> simp

theorem integrable_actual_shiftedGramDeterminant_power (m n d : ℕ) (x : ℝ) :
    Integrable (fun z => (complexSampleGram z+(x : ℂ) • 1).det ^ d)
      (stdGaussian (ComplexSample (m := m) (n := n))) := by
  have h := integrable_actualComplexMatrix_polynomial m n
    (shiftedGramDeterminantPolynomial m n d x)
  simp_rw [eval_shiftedGramDeterminantPolynomial] at h
  exact h

theorem integral_actual_shiftedGramDeterminant_power (m n d : ℕ) (x : ℝ) :
    (∫ z, (complexSampleGram z+(x : ℂ) • 1).det ^ d
      ∂stdGaussian (ComplexSample (m := m) (n := n))) =
        complexGaussianPolynomialMoment (shiftedGramDeterminantPolynomial m n d x) := by
  have h := integral_actualComplexMatrix_polynomial m n
    (shiftedGramDeterminantPolynomial m n d x)
  simp_rw [eval_shiftedGramDeterminantPolynomial] at h
  exact h

end
end JinWishart

def centralGaussianShiftedDeterminantMoment (d n : ℕ) (x : ℝ) : ℝ :=
  (complexGaussianPolynomialMoment (shiftedGramDeterminantPolynomial n n d x)).re

theorem shiftedGramDeterminant_power_real (m n d : ℕ) (x : ℝ) (hx : 0 ≤ x)
    (z : ComplexSample (m := m) (n := n)) :
    (((complexSampleGram z+(x : ℂ) • 1).det.re^d : ℝ) : ℂ) =
      (complexSampleGram z+(x : ℂ) • 1).det^d := by
  have hI : ((x : ℂ) • (1 : Matrix (Fin n) (Fin n) ℂ)).PosSemidef :=
    Matrix.PosSemidef.one.smul (by exact_mod_cast hx)
  have hA := (complexGram_posSemidef (complexSampleMatrix z)).add hI
  have hi := (Complex.nonneg_iff.mp hA.det_nonneg).2
  rw [Complex.ofReal_pow]
  congr 1
  apply Complex.ext
  · simp
  · simpa only [complexSampleGram,Complex.ofReal_im] using hi

theorem integrable_actual_shiftedGramDeterminant_re_power (m n d : ℕ)
    (x : ℝ) (hx : 0 ≤ x) :
    Integrable (fun z => (complexSampleGram z+(x : ℂ) • 1).det.re^d)
      (stdGaussian (ComplexSample (m := m) (n := n))) := by
  have h := (integrable_actual_shiftedGramDeterminant_power m n d x).re
  have he : (fun z : ComplexSample (m := m) (n := n) =>
      (complexSampleGram z+(x : ℂ) • 1).det.re^d) =
        fun z => ((complexSampleGram z+(x : ℂ) • 1).det^d).re := by
    funext z
    rw [← shiftedGramDeterminant_power_real m n d x hx z]
    norm_cast
  rw [he]
  exact h

theorem integral_actual_shiftedGramDeterminant_re_power (d n : ℕ)
    (x : ℝ) (hx : 0 ≤ x) :
    (∫ z, (complexSampleGram z+(x : ℂ) • 1).det.re^d
      ∂stdGaussian (ComplexSample (m := n) (n := n))) =
        centralGaussianShiftedDeterminantMoment d n x := by
  have he : (fun z : ComplexSample (m := n) (n := n) =>
      (complexSampleGram z+(x : ℂ) • 1).det.re^d) =
        fun z => ((complexSampleGram z+(x : ℂ) • 1).det^d).re := by
    funext z
    rw [← shiftedGramDeterminant_power_real n n d x hx z]
    norm_cast
  rw [he]
  calc
    _ = (∫ z, (complexSampleGram z+(x : ℂ) • 1).det^d
        ∂stdGaussian (ComplexSample (m := n) (n := n))).re := by
      exact integral_re (integrable_actual_shiftedGramDeterminant_power n n d x)
    _ = _ := congrArg Complex.re (integral_actual_shiftedGramDeterminant_power n n d x)

theorem centralGaussianShiftedDeterminantMoment_nonneg (d n : ℕ)
    (x : ℝ) (hx : 0 ≤ x) : 0 ≤ centralGaussianShiftedDeterminantMoment d n x := by
  rw [← integral_actual_shiftedGramDeterminant_re_power d n x hx]
  apply integral_nonneg
  intro z
  have hI : ((x : ℂ) • (1 : Matrix (Fin n) (Fin n) ℂ)).PosSemidef :=
    Matrix.PosSemidef.one.smul (by exact_mod_cast hx)
  exact pow_nonneg (Complex.nonneg_iff.mp
    ((complexGram_posSemidef (complexSampleMatrix z)).add hI).det_nonneg).1 d

theorem lintegral_actualCentralSquare_shiftedDeterminant_power (d n : ℕ)
    (x : ℝ) (hx : 0 ≤ x) :
    (∫⁻ A, ENNReal.ofReal ((A+(x : ℂ) • 1).det.re^d)
      ∂complexWishart (m := n) (n := n)) =
        ENNReal.ofReal (centralGaussianShiftedDeterminantMoment d n x) := by
  unfold complexWishart
  rw [lintegral_map (by fun_prop) (by fun_prop)]
  rw [← integral_actual_shiftedGramDeterminant_re_power d n x hx,
    ofReal_integral_eq_lintegral_ofReal
      (integrable_actual_shiftedGramDeterminant_re_power n n d x hx)]
  exact ae_of_all _ (fun z => by
    have hI : ((x : ℂ) • (1 : Matrix (Fin n) (Fin n) ℂ)).PosSemidef :=
      Matrix.PosSemidef.one.smul (by exact_mod_cast hx)
    exact pow_nonneg (Complex.nonneg_iff.mp
      ((complexGram_posSemidef (complexSampleMatrix z)).add hI).det_nonneg).1 d)

end
end JinWishart

def centralRectangularMinimumPolynomial (d n : ℕ) (x : ℝ) : ℝ :=
  (centralDeterminantPowerMoment 0 n d)⁻¹ * centralGaussianShiftedDeterminantMoment d n x

theorem centralRectangularMinimumPolynomial_nonneg (d n : ℕ)
    (x : ℝ) (hx : 0 ≤ x) : 0 ≤ centralRectangularMinimumPolynomial d n x :=
  mul_nonneg (inv_nonneg.mpr (centralDeterminantPowerMoment_pos 0 n d).le)
    (centralGaussianShiftedDeterminantMoment_nonneg d n x hx)

theorem centralWishartDensityConstant_eq_square_ratio (d n : ℕ) :
    centralWishartDensityConstant d n =
      centralWishartDensityConstant 0 n * (centralDeterminantPowerMoment 0 n d)⁻¹ := by
  have h := centralWishartDensityConstant_power_bias 0 n d
  simp only [Nat.zero_add] at h
  have hM := (centralDeterminantPowerMoment_pos 0 n d).ne'
  apply (eq_mul_inv_iff_mul_eq₀ hM).mpr
  nlinarith

theorem actualCentralRectangular_shiftedCone_probability (d n : ℕ)
    (x : ℝ) (hx : 0 ≤ x) :
    ((complexWishart (m := d+n) (n := n)).map realCoordinatesOfHermitian)
      (centralCoordinateShiftedCone n x) =
        ENNReal.ofReal (Real.exp (-(n : ℝ)*x) * centralRectangularMinimumPolynomial d n x) := by
  have hp := actualCentralWishart_shiftedCone_probability d n x hx
  have hm := lintegral_complexWishart_eq_cone 0 n
    (fun A => ENNReal.ofReal ((A+(x : ℂ) • 1).det.re^d)) (by fun_prop)
  simp only [Nat.zero_add,centralConeWeight,pow_zero,one_mul] at hm
  simp_rw [ENNReal.ofReal_mul' (Real.exp_nonneg _)] at hp
  rw [centralWishartDensityConstant_eq_square_ratio,
    ENNReal.ofReal_mul (centralWishartDensityConstant_pos 0 n).le] at hp
  rw [hp,ENNReal.ofReal_mul (Real.exp_nonneg _)]
  calc
    _ = ENNReal.ofReal (Real.exp (-(n : ℝ)*x)) *
        (ENNReal.ofReal ((centralDeterminantPowerMoment 0 n d)⁻¹) *
          (∫⁻ A, ENNReal.ofReal ((A+(x : ℂ) • 1).det.re^d)
            ∂complexWishart (m := n) (n := n))) := by rw [hm]; ring
    _ = _ := by
      rw [lintegral_actualCentralSquare_shiftedDeterminant_power d n x hx,
        centralRectangularMinimumPolynomial,
        ENNReal.ofReal_mul (inv_nonneg.mpr (centralDeterminantPowerMoment_pos 0 n d).le)]

theorem actualCentralRectangular_minimum_CDF (d n : ℕ) (hn : 0 < n)

    ((stdGaussian (ComplexSample (m := d+n) (n := n)))

theorem actualCentralRectangular_minimum_cdf_function (d n : ℕ) (hn : 0 < n) (x : ℝ) :
    cdf ((stdGaussian (ComplexSample (m := d+n) (n := n))).map
      (centralGaussianSampleMinimum (d+n) n hn)) x =
      if 0 ≤ x then 1-Real.exp (-(n : ℝ)*x) * centralRectangularMinimumPolynomial d n x else 0 := by
  rw [actualCentralGaussian_minimum_cdf_eq_event]
  by_cases hx : 0 ≤ x
  · rw [if_pos hx]
