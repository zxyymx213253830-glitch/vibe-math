theorem complexGaussianPolynomialMoment_monomial {ι : Type*} [Fintype ι]
    (d : (ι ⊕ ι) →₀ ℕ) (c : ℂ) :
    complexGaussianPolynomialMoment (MvPolynomial.monomial d c) =
      c * ∏ i, if d (Sum.inl i)=d (Sum.inr i) then (Nat.factorial (d (Sum.inl i)) : ℂ) else 0 := by
  classical
  by_cases hc : c=0
  · simp [hc,complexGaussianPolynomialMoment]
  · simp [complexGaussianPolynomialMoment,MvPolynomial.support_monomial,hc]

theorem complexGaussianPolynomialMoment_X_conj_X {ι : Type*} [Fintype ι] [DecidableEq ι]
    (a b : ι) :
    complexGaussianPolynomialMoment
      (MvPolynomial.X (Sum.inl a)*MvPolynomial.X (Sum.inr b) : MvPolynomial (ι ⊕ ι) ℂ) =
      if a=b then 1 else 0 := by
  classical
  simp only [MvPolynomial.X,MvPolynomial.monomial_mul_monomial,one_mul]
  rw [complexGaussianPolynomialMoment_monomial]
  simp only [one_mul,Finsupp.add_apply,Finsupp.single_apply,Sum.inl.injEq,Sum.inr.injEq,
    Sum.inl_ne_inr,Sum.inr_ne_inl,ite_false,zero_add,add_zero]
  by_cases hab : a=b
  · subst b
    simp only [ite_true]
    apply Finset.prod_eq_one
    intro i _
    split_ifs <;> norm_num
  · rw [if_neg hab]
    apply Finset.prod_eq_zero (Finset.mem_univ a)
    simp [hab,Ne.symm hab]

theorem complexGaussianPolynomialMoment_monomial_unbalanced {ι : Type*} [Fintype ι]
    (d : (ι ⊕ ι) →₀ ℕ) (c : ℂ)
    (hne : (∑ i, d (Sum.inl i)) ≠ ∑ i, d (Sum.inr i)) :
    complexGaussianPolynomialMoment (MvPolynomial.monomial d c) = 0 := by
  classical
  rw [complexGaussianPolynomialMoment_monomial]
  have hp : (∏ i, if d (Sum.inl i)=d (Sum.inr i)
      then (Nat.factorial (d (Sum.inl i)) : ℂ) else 0) = 0 := by
    by_contra hp
    apply hne
    apply Finset.sum_congr rfl
    intro i _
    by_contra hi
    exact hp (Finset.prod_eq_zero (Finset.mem_univ i) (by simp [hi]))
  rw [hp,mul_zero]

theorem complexGaussianPolynomialMoment_fourth {ι : Type*} [Fintype ι] [DecidableEq ι]
    (a b c d : ι) :
    complexGaussianPolynomialMoment
      (MvPolynomial.X (Sum.inl a)*MvPolynomial.X (Sum.inl b)*
        MvPolynomial.X (Sum.inr c)*MvPolynomial.X (Sum.inr d) : MvPolynomial (ι ⊕ ι) ℂ) =
      (if a=c ∧ b=d then 1 else 0)+(if a=d ∧ b=c then 1 else 0) := by
  classical
  have hw (a b c d : ι) :
      complexGaussianPolynomialMoment
        (MvPolynomial.X (Sum.inl a)*MvPolynomial.X (Sum.inl b)*
          MvPolynomial.X (Sum.inr c)*MvPolynomial.X (Sum.inr d) : MvPolynomial (ι ⊕ ι) ℂ) =
        ∏ i, if (if a=i then 1 else 0)+(if b=i then 1 else 0) =
          ((if c=i then 1 else 0)+(if d=i then 1 else 0) : ℕ)
          then (Nat.factorial ((if a=i then 1 else 0)+(if b=i then 1 else 0)) : ℂ) else 0 := by
    simp only [MvPolynomial.X,MvPolynomial.monomial_mul_monomial,one_mul]
    rw [complexGaussianPolynomialMoment_monomial]
    simp only [one_mul,Finsupp.add_apply,Finsupp.single_apply,Sum.inl.injEq,Sum.inr.injEq,
      Sum.inl_ne_inr,Sum.inr_ne_inl,ite_false,zero_add,add_zero]
  have hm (a b : ι) : (∏ i : ι,
      (Nat.factorial ((if a=i then 1 else 0)+(if b=i then 1 else 0)) : ℂ)) =
      if a=b then 2 else 1 := by
    by_cases hab : a=b
    · subst b
      rw [if_pos rfl,Finset.prod_eq_single a]
      · norm_num
      · intro i _ hia
        simp [Ne.symm hia]
      · simp
    · rw [if_neg hab]
      apply Finset.prod_eq_one
      intro i _
      split_ifs <;> simp_all
  by_cases hp : (a=c ∧ b=d) ∨ (a=d ∧ b=c)
  · rcases hp with ⟨rfl,rfl⟩ | ⟨rfl,rfl⟩
    · rw [hw]
      simp only [ite_true]
      rw [hm]
      by_cases hab : a=b
      · subst b; norm_num
      · simp [hab,Ne.symm hab]
    · rw [hw]
      have he (i : ι) :
          (if a=i then 1 else 0)+(if b=i then 1 else 0) =
            ((if b=i then 1 else 0)+(if a=i then 1 else 0) : ℕ) := Nat.add_comm _ _
      simp only [he,ite_true]
      rw [hm]
      by_cases hab : a=b
      · subst b; norm_num
      · simp [hab,Ne.symm hab]
  · have h1 : ¬(a=c ∧ b=d) := fun h => hp (Or.inl h)
    have h2 : ¬(a=d ∧ b=c) := fun h => hp (Or.inr h)
    rw [if_neg h1,if_neg h2,add_zero,hw]
    by_contra hne
    have hc (i : ι) : (if a=i then 1 else 0)+(if b=i then 1 else 0) =
        ((if c=i then 1 else 0)+(if d=i then 1 else 0) : ℕ) := by
      by_contra hi
      exact hne (Finset.prod_eq_zero (Finset.mem_univ i) (by simp [hi]))
    apply hp
    by_cases hab : a=b
    · subst b
      have ha := hc a
      by_cases hca : c=a
      · by_cases hda : d=a
        · exact Or.inl ⟨hca.symm,hda.symm⟩
        · simp [hca,hda] at ha
      · by_cases hda : d=a
        · simp [hca,hda] at ha
        · simp [hca,hda] at ha
    · by_cases hac : a=c
      · subst c
        have hb := hc b
        by_cases hbd : b=d
        · exact Or.inl ⟨rfl,hbd⟩
        · simp [hab,Ne.symm hab,Ne.symm hbd] at hb
      · have ha := hc a
        by_cases had : a=d
        · subst d
          have hb := hc b
          by_cases hbc : b=c
          · exact Or.inr ⟨rfl,hbc⟩
          · simp [hab,Ne.symm hab,Ne.symm hbc] at hb
        · simp [hab,hac,had,Ne.symm hab,Ne.symm hac,Ne.symm had] at ha

end

theorem complexGaussianPolynomialMoment_shifted_fourth {ι : Type*} [Fintype ι] [DecidableEq ι]
    (a b c d : ι) (α β γ δ : ℂ) :
    complexGaussianPolynomialMoment
      ((MvPolynomial.X (Sum.inl a)+MvPolynomial.C α)*
       (MvPolynomial.X (Sum.inl b)+MvPolynomial.C β)*
       (MvPolynomial.X (Sum.inr c)+MvPolynomial.C γ)*
       (MvPolynomial.X (Sum.inr d)+MvPolynomial.C δ) : MvPolynomial (ι ⊕ ι) ℂ) =
      α*β*γ*δ + (if a=c then 1 else 0)*β*δ + (if a=d then 1 else 0)*β*γ +
      (if b=c then 1 else 0)*α*δ + (if b=d then 1 else 0)*α*γ +
      (if a=c ∧ b=d then 1 else 0)+(if a=d ∧ b=c then 1 else 0) := by
  classical
  let U : MvPolynomial (ι ⊕ ι) ℂ := MvPolynomial.X (Sum.inl a)
  let V : MvPolynomial (ι ⊕ ι) ℂ := MvPolynomial.X (Sum.inl b)
  let W : MvPolynomial (ι ⊕ ι) ℂ := MvPolynomial.X (Sum.inr c)
  let Y : MvPolynomial (ι ⊕ ι) ℂ := MvPolynomial.X (Sum.inr d)
  have he : (U+MvPolynomial.C α)*(V+MvPolynomial.C β)*(W+MvPolynomial.C γ)*(Y+MvPolynomial.C δ) =
      U*V*W*Y + δ • (U*V*W) + γ • (U*V*Y) + β • (U*W*Y) + α • (V*W*Y) +
      (γ*δ) • (U*V) + (β*δ) • (U*W) + (β*γ) • (U*Y) +
      (α*δ) • (V*W) + (α*γ) • (V*Y) + (α*β) • (W*Y) +
      (β*γ*δ) • U + (α*γ*δ) • V + (α*β*δ) • W + (α*β*γ) • Y + MvPolynomial.C (α*β*γ*δ) := by
    simp only [MvPolynomial.smul_eq_C_mul,map_mul]
    ring
  have hL (P : MvPolynomial (ι ⊕ ι) ℂ) :
      complexGaussianMomentLinear P = complexGaussianPolynomialMoment P := rfl
  change complexGaussianMomentLinear
    ((U+MvPolynomial.C α)*(V+MvPolynomial.C β)*(W+MvPolynomial.C γ)*(Y+MvPolynomial.C δ)) = _
  rw [he]
  simp only [map_add,map_smul]
  simp only [hL]
  simp only [U,V,W,Y,complexGaussianPolynomialMoment_fourth,
    complexGaussianPolynomialMoment_direct_direct_conjugate,
    complexGaussianPolynomialMoment_direct_conjugate_conjugate,
    complexGaussianPolynomialMoment_direct_pair,complexGaussianPolynomialMoment_conjugate_pair,
    complexGaussianPolynomialMoment_X_conj_X,complexGaussianPolynomialMoment_X,
    complexGaussianPolynomialMoment_C,smul_eq_mul,mul_zero,zero_mul,add_zero,zero_add]
  ring

end
