  have h2 := mul_le_mul_of_nonneg_left h.2 hr.le
  have he : r*((r*x)⁻¹-Real.log (1+(r*x)⁻¹))=x⁻¹-r*Real.log (1+(r*x)⁻¹) := by
    field_simp [hr.ne',hx.ne'] <;> ring

  refine ⟨h0,?_⟩
  convert h2 using 1 <;> field_simp [hr.ne',hx.ne'] <;> ring
