theorem integrable_log_of_integrable_inverse_rpow {α : Type*} [MeasurableSpace α]
    (μ : Measure α) (f : α → ℝ) (hf : Integrable f μ)
    (hfpos : ∀ᵐ a ∂μ, 0 < f a) (p : ℝ) (hp : 0 < p)
    (hinv : Integrable (fun a => ((f a)⁻¹)^p) μ) :
    Integrable (fun a => Real.log (f a)) μ := by
  apply (hf.add (hinv.div_const p)).mono'
    (Real.measurable_log.comp_aemeasurable hf.aemeasurable).aestronglyMeasurable
  filter_upwards [hfpos] with a ha
  change ‖Real.log (f a)‖ ≤ f a+((f a)⁻¹)^p/p
  rw [Real.norm_eq_abs]
  have hpow : 0 ≤ ((f a)⁻¹)^p/p := div_nonneg (Real.rpow_nonneg (by positivity) _) hp.le
  have hu := Real.log_le_sub_one_of_pos ha
  have hl := Real.log_le_rpow_div (inv_nonneg.mpr ha.le) hp
  rw [Real.log_inv] at hl
  exact abs_le.mpr ⟨by linarith,by linarith⟩

theorem integrable_log_one_add_mul {α : Type*} [MeasurableSpace α]
    (μ : Measure α) (f : α → ℝ) (hf : Integrable f μ)
    (hfnonneg : ∀ᵐ a ∂μ, 0 ≤ f a) (r : ℝ) (hr : 0 ≤ r) :
    Integrable (fun a => Real.log (1+r*f a)) μ := by
  apply (hf.const_mul r).mono'
    (Real.measurable_log.comp_aemeasurable
      (aemeasurable_const.add (aemeasurable_const.mul hf.aemeasurable))).aestronglyMeasurable
  filter_upwards [hfnonneg] with a ha
  change ‖Real.log (1+r*f a)‖ ≤ r*f a
  have hpos : 0 < 1+r*f a := by positivity
  rw [Real.norm_eq_abs,abs_of_nonneg (Real.log_nonneg (by nlinarith))]
  have h := Real.log_le_sub_one_of_pos hpos
  linarith

theorem integral_log_add_inverse_tendsto {α : Type*} [MeasurableSpace α]
    (μ : Measure α) (f : α → ℝ) (hf : Integrable f μ)
    (hfpos : ∀ᵐ a ∂μ, 0 < f a) (hlog : Integrable (fun a => Real.log (f a)) μ) :
    Tendsto (fun r : ℝ => ∫ a, Real.log (f a+r⁻¹) ∂μ) atTop
      (𝓝 (∫ a, Real.log (f a) ∂μ)) := by
  apply tendsto_integral_filter_of_dominated_convergence
    (fun a => |Real.log (f a)|+f a)
  · exact Filter.Eventually.of_forall (fun r =>
      (Real.measurable_log.comp_aemeasurable (hf.aemeasurable.add_const _)).aestronglyMeasurable)
  · filter_upwards [eventually_ge_atTop (1 : ℝ)] with r hr
    filter_upwards [hfpos] with a ha
    have hrp : 0 < r := lt_of_lt_of_le zero_lt_one hr
    have hi : r⁻¹ ≤ 1 := by
      simpa using one_div_le_one_div_of_le zero_lt_one hr
    have hpos : 0 < f a+r⁻¹ := by positivity
    have hl := Real.log_le_log ha (le_add_of_nonneg_right (inv_nonneg.mpr hrp.le))
    have hu := (Real.log_le_sub_one_of_pos hpos)
    rw [Real.norm_eq_abs]
    exact abs_le.mpr ⟨by have := neg_abs_le (Real.log (f a)); linarith,
      by have := le_abs_self (Real.log (f a)); have := abs_nonneg (Real.log (f a)); linarith⟩
  · exact hlog.abs.add hf
  · filter_upwards [hfpos] with a ha
    have ht : Tendsto (fun r : ℝ => f a+r⁻¹) atTop (𝓝 (f a)) := by
      simpa using (tendsto_const_nhds.add tendsto_inv_atTop_zero :
        Tendsto (fun r : ℝ => f a+r⁻¹) atTop (𝓝 (f a+0)))
    exact (Real.continuousAt_log ha.ne').tendsto.comp ht

end

def finiteSpectrumLogDet {ι : Type*} [Fintype ι] (l : ι → ℝ) (r : ℝ) : ℝ :=
  ∑ i, Real.log (1+r*l i)

theorem integrable_finiteSpectrumLogDet {α ι : Type*} [MeasurableSpace α] [Fintype ι]
    (μ : Measure α) (f : ι → α → ℝ) (hf : ∀ i, Integrable (f i) μ)
    (hpos : ∀ i, ∀ᵐ a ∂μ, 0 < f i a) (r : ℝ) (hr : 0 ≤ r) :
    Integrable (fun a => finiteSpectrumLogDet (fun i => f i a) r) μ := by
  exact integrable_finsetSum _ (fun i _ =>
    integrable_log_one_add_mul μ (f i) (hf i) ((hpos i).mono (fun _ h => h.le)) r hr)

theorem integral_finiteSpectrumLogDet_offset_tendsto {α ι : Type*}
    [MeasurableSpace α] [Fintype ι] (μ : Measure α) [IsProbabilityMeasure μ]
    (f : ι → α → ℝ) (hf : ∀ i, Integrable (f i) μ)
    (hpos : ∀ i, ∀ᵐ a ∂μ, 0 < f i a)
    (hlog : ∀ i, Integrable (fun a => Real.log (f i a)) μ) :
    Tendsto (fun r : ℝ => (∫ a, finiteSpectrumLogDet (fun i => f i a) r ∂μ)-
      (Fintype.card ι : ℝ)*Real.log r) atTop
      (𝓝 (∑ i, ∫ a, Real.log (f i a) ∂μ)) := by
  classical
  have ht := tendsto_finsetSum Finset.univ (fun i _ =>
    integral_log_add_inverse_tendsto μ (f i) (hf i) (hpos i) (hlog i))
  apply ht.congr'
  filter_upwards [eventually_ge_atTop (1 : ℝ)] with r hr
  have hrp : 0 < r := lt_of_lt_of_le zero_lt_one hr
  have hi (i : ι) := integrable_log_one_add_mul μ (f i) (hf i)
    ((hpos i).mono (fun _ h => h.le)) r hrp.le
  have he (i : ι) : (∫ a, Real.log (1+r*f i a) ∂μ)-Real.log r =
      ∫ a, Real.log (f i a+r⁻¹) ∂μ := by
    calc
      _ = ∫ a, (Real.log (1+r*f i a)-Real.log r) ∂μ := by
        rw [integral_sub (hi i) (integrable_const _)]
        simp
      _ = _ := integral_congr_ae ((hpos i).mono (fun a ha => by
        have hp : 0 < f i a+r⁻¹ := by positivity
        change Real.log (1+r*f i a)-Real.log r = Real.log (f i a+r⁻¹)
        rw [show 1+r*f i a=r*(f i a+r⁻¹) by field_simp [hrp.ne']; ring,
          Real.log_mul hrp.ne' hp.ne']
        ring))
  rw [show (∫ a, finiteSpectrumLogDet (fun i => f i a) r ∂μ)=
      ∑ i, ∫ a, Real.log (1+r*f i a) ∂μ from integral_finsetSum _ (fun i _ => hi i)]
  simp_rw [← he]
  simp [Finset.sum_sub_distrib]

theorem integral_finiteSpectrumLogDet_prelog {α ι : Type*}
    [MeasurableSpace α] [Fintype ι] (μ : Measure α) [IsProbabilityMeasure μ]
    (f : ι → α → ℝ) (hf : ∀ i, Integrable (f i) μ)
    (hpos : ∀ i, ∀ᵐ a ∂μ, 0 < f i a)
    (hlog : ∀ i, Integrable (fun a => Real.log (f i a)) μ) :
    Tendsto (fun r : ℝ => (∫ a, finiteSpectrumLogDet (fun i => f i a) r ∂μ)/Real.log r)

    atTop (𝓝 (Fintype.card ι : ℝ))).add ht
  apply (show Tendsto (fun r : ℝ => (Fintype.card ι : ℝ)+
      ((∫ a, finiteSpectrumLogDet (fun i => f i a) r ∂μ)-
        (Fintype.card ι : ℝ)*Real.log r)/Real.log r) atTop
      (𝓝 (Fintype.card ι : ℝ)) by simpa using hs).congr'
  filter_upwards [eventually_gt_atTop (1 : ℝ)] with r hr
  have hlogr := (Real.log_pos hr).ne'
  field_simp
  <;> ring

end

theorem hermitian_det_one_add_smul_eigenvalues {n : ℕ}
    (A : Matrix (Fin n) (Fin n) ℂ) (hA : A.IsHermitian) (r : ℝ) :
    (1+(r : ℂ) • A).det = ((∏ i, (1+r*hA.eigenvalues₀ i) : ℝ) : ℂ) := by
  classical
  let U := hA.eigenvectorUnitary
  let D : Matrix (Fin n) (Fin n) ℂ := diagonal (RCLike.ofReal ∘ hA.eigenvalues)
  have he : 1+(r : ℂ) • A = Unitary.conjStarAlgAut ℂ _ U (1+(r : ℂ) • D) := by
    rw [map_add,map_one,map_smul]
    rw [← hA.spectral_theorem]
  have hu := congrArg Matrix.det (Unitary.coe_mul_star_self U)
  simp only [Matrix.det_mul,Matrix.det_one] at hu
  rw [Unitary.coe_star] at hu
  have hd : (1+(r : ℂ) • D).det = ((∏ i, (1+r*hA.eigenvalues i) : ℝ) : ℂ) := by
    have hdiag : 1+(r : ℂ) • D = diagonal (fun i => ((1+r*hA.eigenvalues i : ℝ) : ℂ)) := by
      ext i j
      by_cases hij : i=j
      · subst j
        simp [D,Matrix.add_apply,Matrix.smul_apply,Matrix.diagonal_apply]
      · simp [D,Matrix.add_apply,Matrix.smul_apply,Matrix.diagonal_apply,Matrix.one_apply,hij]
    rw [hdiag,Matrix.det_diagonal]
    norm_cast
  rw [he,Unitary.conjStarAlgAut_apply,Matrix.det_mul,Matrix.det_mul]
  calc
    _ = (1+(r : ℂ) • D).det * ((U : Matrix (Fin n) (Fin n) ℂ).det*
        (star (U : Matrix (Fin n) (Fin n) ℂ)).det) := by ring
    _ = (1+(r : ℂ) • D).det := by rw [hu,mul_one]
    _ = _ := by
      rw [hd]
      congr 1
      exact Equiv.prod_comp (Fintype.equivOfCardEq (Fintype.card_fin (Fintype.card (Fin n)))).symm
        (fun i => 1+r*hA.eigenvalues₀ i)

theorem gram_real_logdet_eq_spectral {m n : ℕ} (X : Matrix (Fin m) (Fin n) ℂ)
    (r : ℝ) (hr : 0 ≤ r) :
    Real.log ((1+(r : ℂ) • complexGram X).det.re) =
      finiteSpectrumLogDet (complexGram_posSemidef X).1.eigenvalues₀ r := by
  rw [hermitian_det_one_add_smul_eigenvalues _ (complexGram_posSemidef X).1]
  simp only [Complex.ofReal_re,finiteSpectrumLogDet]
  exact Real.log_prod (fun i _ => (show 0 < 1+r*(complexGram_posSemidef X).1.eigenvalues₀ i by
    have := complexGram_eigenvalues₀_nonneg X i
    positivity).ne')

theorem hermitian_det_eq_prod_ordered {n : ℕ}
    (A : Matrix (Fin n) (Fin n) ℂ) (hA : A.IsHermitian) :
    A.det = ((∏ i, hA.eigenvalues₀ i : ℝ) : ℂ) := by
  classical
  rw [hA.det_eq_prod_eigenvalues]
  push_cast
  exact Equiv.prod_comp (Fintype.equivOfCardEq (Fintype.card_fin (Fintype.card (Fin n)))).symm
    (fun i => (hA.eigenvalues₀ i : ℂ))

theorem hermitian_real_logdet_eq_sum_log {n : ℕ}
    (A : Matrix (Fin n) (Fin n) ℂ) (hA : A.IsHermitian)
    (hpos : ∀ i, 0 < hA.eigenvalues₀ i) :
    Real.log A.det.re = ∑ i, Real.log (hA.eigenvalues₀ i) := by
  rw [hermitian_det_eq_prod_ordered A hA]
  simp only [Complex.ofReal_re]
  exact Real.log_prod (fun i _ => (hpos i).ne')

end
