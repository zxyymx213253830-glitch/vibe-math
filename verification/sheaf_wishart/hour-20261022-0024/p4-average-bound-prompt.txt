theorem rectangular_det_permutation_weight_sum {q m : ℕ}
    (A : Matrix (Fin q) (Fin m) ℂ) (B : Matrix (Fin m) (Fin q) ℂ)
    (σ : Equiv.Perm (Fin q)) :
    (∑ f : Fin q → Fin m, (∏ i, A (σ i) (f i))*(B.submatrix f id).det) =
      ((Equiv.Perm.sign σ : ℤ) : ℂ)*(A*B).det := by
  classical
  let E : (Fin q → Fin m) ≃ (Fin q → Fin m) :=
    { toFun := fun f i => f (σ i)
      invFun := fun f i => f (σ.symm i)
      left_inv := by intro f; funext i; simp
      right_inv := by intro f; funext i; simp }
  calc
    _ = ∑ g : Fin q → Fin m, (∏ i, A (σ i) (E g i))*(B.submatrix (E g) id).det :=
      (E.sum_comp (fun f => (∏ i, A (σ i) (f i))*(B.submatrix f id).det)).symm
    _ = ∑ g : Fin q → Fin m, ((Equiv.Perm.sign σ : ℤ) : ℂ)*
        ((∏ i, A i (g i))*(B.submatrix g id).det) := by
      apply Finset.sum_congr rfl
      intro g hg
      have hp : (∏ i, A (σ i) (E g i)) = ∏ i, A i (g i) :=
        Equiv.prod_comp σ (fun i => A i (g i))
      have he : B.submatrix (E g) id = (B.submatrix g id).submatrix σ id := by
        ext i j
        rfl
      rw [hp,he,Matrix.det_permute]
      ring
    _ = _ := by rw [← Finset.mul_sum,← det_mul_rectangular_sum_functions]

theorem det_mul_rectangular_factorial_sum_functions {q m : ℕ}
    (A : Matrix (Fin q) (Fin m) ℂ) (B : Matrix (Fin m) (Fin q) ℂ) :
    (q.factorial : ℂ)*(A*B).det =
      ∑ f : Fin q → Fin m, (A.submatrix id f).det*(B.submatrix f id).det := by
  classical
  have ha (f : Fin q → Fin m) : (A.submatrix id f).det =
      ∑ σ : Equiv.Perm (Fin q), ((Equiv.Perm.sign σ : ℤ) : ℂ)*(∏ i, A (σ i) (f i)) := by
    rw [Matrix.det_apply']
    rfl
  symm
  calc
    _ = ∑ σ : Equiv.Perm (Fin q), ((Equiv.Perm.sign σ : ℤ) : ℂ)*
        (∑ f : Fin q → Fin m, (∏ i, A (σ i) (f i))*(B.submatrix f id).det) := by
      simp_rw [ha,Finset.sum_mul]
      rw [Finset.sum_comm]
      apply Finset.sum_congr rfl
      intro σ hσ
      rw [Finset.mul_sum]
      apply Finset.sum_congr rfl
      intro f hf
      ring
    _ = ∑ _σ : Equiv.Perm (Fin q), (A*B).det := by
      apply Finset.sum_congr rfl
      intro σ hσ
      rw [rectangular_det_permutation_weight_sum]
      have hs : (((Equiv.Perm.sign σ : ℤ) : ℂ))*(((Equiv.Perm.sign σ : ℤ) : ℂ))=1 := by
        exact_mod_cast Int.isUnit_mul_self (Equiv.Perm.sign σ).isUnit
      rw [← mul_assoc,hs,one_mul]
    _ = _ := by simp [Fintype.card_perm]

theorem gram_det_factorial_sum_squared_minors {m q : ℕ}
    (B : Matrix (Fin m) (Fin q) ℂ) :
    (q.factorial : ℂ)*(complexGram B).det =
      ∑ f : Fin q → Fin m, star ((B.submatrix f id).det)*(B.submatrix f id).det := by
  classical
  have h := det_mul_rectangular_factorial_sum_functions B.conjTranspose B
  have he (f : Fin q → Fin m) : B.conjTranspose.submatrix id f = (B.submatrix f id).conjTranspose := by
    ext i j
    rfl
  simpa only [he,Matrix.det_conjTranspose,complexGram] using h

theorem squared_minor_le_factorial_gram_det {m q : ℕ}
    (B : Matrix (Fin m) (Fin q) ℂ) (f : Fin q → Fin m) :
    Complex.normSq ((B.submatrix f id).det) ≤ (q.factorial : ℝ)*(complexGram B).det.re := by
  classical
  have h := congrArg Complex.re (gram_det_factorial_sum_squared_minors B)
  have ht (z : ℂ) : (star z*z).re = Complex.normSq z := by
    simp [Complex.mul_re,Complex.normSq_apply,Complex.star_def]
  have hr : (q.factorial : ℝ)*(complexGram B).det.re =
      ∑ g : Fin q → Fin m, Complex.normSq ((B.submatrix g id).det) := by
    simpa [Complex.mul_re,Complex.normSq_apply] using h
  rw [hr]
  exact Finset.single_le_sum (fun g _ => Complex.normSq_nonneg ((B.submatrix g id).det))
    (Finset.mem_univ f)

end

theorem selected_gram_det_le_coefficient {m n q : ℕ}
    (X : Matrix (Fin m) (Fin n) ℂ) (g : Fin q → Fin n) (hg : Injective g) :
    (complexGram (X.submatrix id g)).det.re ≤
      (((1+(Polynomial.X : ℂ[X]) • (complexGram X).map Polynomial.C).det).coeff q).re := by
  classical
  let S := Finset.univ.image g
  let f : Fin q → S := fun i => ⟨g i,Finset.mem_image_of_mem g (Finset.mem_univ i)⟩
  have hf : Bijective f := by
    constructor
    · intro i j hij
      exact hg (congrArg Subtype.val hij)
    · intro j
      obtain ⟨i,hi,hij⟩ := Finset.mem_image.mp j.property
      exact ⟨i,Subtype.ext hij⟩
  let e : Fin q ≃ S := Equiv.ofBijective f hf
  have he : ((complexGram X).submatrix (Subtype.val : S → Fin n) (Subtype.val : S → Fin n)).submatrix e e =
      complexGram (X.submatrix id g) := by
    ext i j
    rfl
  have hd : ((complexGram X).submatrix (Subtype.val : S → Fin n) (Subtype.val : S → Fin n)).det =
      (complexGram (X.submatrix id g)).det := by
    rw [← he,Matrix.det_submatrix_equiv_self]
  have hS : S ∈ Finset.univ.powersetCard q := by
    simp [S,Finset.mem_powersetCard,Finset.card_image_of_injective _ hg]
  have hcoeff : (((1+(Polynomial.X : ℂ[X]) • (complexGram X).map Polynomial.C).det).coeff q).re =
      ∑ s ∈ Finset.univ.powersetCard q,
        ((complexGram X).submatrix (Subtype.val : s → Fin n) (Subtype.val : s → Fin n)).det.re := by
    simpa using congrArg Complex.re (Matrix.coeff_det_one_add_X_smul_eq_sum_minors (complexGram X) q)
  rw [hcoeff,← hd]
  refine Finset.single_le_sum (s := Finset.univ.powersetCard q)
    (f := fun s : Finset (Fin n) =>
      ((complexGram X).submatrix (Subtype.val : s → Fin n) (Subtype.val : s → Fin n)).det.re) ?_ hS
  intro s hs
  exact (RCLike.nonneg_iff.mp
    (((complexGram_posSemidef X).submatrix (Subtype.val : s → Fin n)).det_nonneg)).1

theorem squared_coordinate_minor_le_factorial_coefficient {m n q : ℕ}
    (X : Matrix (Fin m) (Fin n) ℂ) (f : Fin q → Fin m) (g : Fin q → Fin n)
    (hg : Injective g) :
    Complex.normSq ((X.submatrix f g).det) ≤ (q.factorial : ℝ)*
      (((1+(Polynomial.X : ℂ[X]) • (complexGram X).map Polynomial.C).det).coeff q).re := by
  have h := squared_minor_le_factorial_gram_det (X.submatrix id g) f
  have he : (X.submatrix id g).submatrix f id = X.submatrix f g := rfl
  rw [he] at h
  exact h.trans (mul_le_mul_of_nonneg_left (selected_gram_det_le_coefficient X g hg) (Nat.cast_nonneg _))

end
end JinWishart

theorem gram_rank_coefficient_log_eq_ordered_positive_sum {m n : ℕ}
    (X : Matrix (Fin m) (Fin n) ℂ) :
    Real.log (((1+(Polynomial.X : ℂ[X]) • (complexGram X).map Polynomial.C).det).coeff X.rank).re =
      ∑ i, positiveSpectrumLog ((complexGram_posSemidef X).1.eigenvalues₀ i) := by
  classical
  let h := (complexGram_posSemidef X).1
  let e : Fin (Fintype.card (Fin n)) ≃ Fin n :=
    Fintype.equivOfCardEq (Fintype.card_fin (Fintype.card (Fin n)))
  have hs : (∑ i, positiveSpectrumLog (h.eigenvalues₀ i)) =
      ∑ i, positiveSpectrumLog (h.eigenvalues i) := by
    simpa only [Matrix.IsHermitian.eigenvalues,e,Equiv.symm_apply_apply] using
      Equiv.sum_comp e (fun i => positiveSpectrumLog (h.eigenvalues i))
  rw [gram_rank_coefficient_log_eq_positive_sum]
  simpa only [positiveSpectrumLog] using hs.symm

theorem abs_positive_log_sum_le_minor_log {m n q : ℕ}
    (X : Matrix (Fin m) (Fin n) ℂ) (f : Fin q → Fin m) (g : Fin q → Fin n)
    (hg : Function.Injective g) (hr : X.rank=q) (hp : (X.submatrix f g).det≠0) :
    |∑ i, positiveSpectrumLog ((complexGram_posSemidef X).1.eigenvalues₀ i)| ≤
      (∑ i, (complexGram_posSemidef X).1.eigenvalues₀ i)+|Real.log (q.factorial : ℝ)|+
        2*|Real.log ‖(X.submatrix f g).det‖| := by
  classical
  have hb := squared_coordinate_minor_le_factorial_coefficient X f g hg
  have hc : 0 < (((1+(Polynomial.X : ℂ[X]) • (complexGram X).map Polynomial.C).det).coeff q).re := by
    rw [← hr]
    exact gram_rank_coefficient_re_pos X
  have hf : 0 < (q.factorial : ℝ) := Nat.cast_pos.mpr (Nat.factorial_pos q)
  have hn : 0 < Complex.normSq ((X.submatrix f g).det) := by
    rw [Complex.normSq_eq_norm_sq]
    exact pow_pos (norm_pos_iff.mpr hp) 2
  have hl := (Real.log_le_log_iff hn (mul_pos hf hc)).2 hb
  rw [Complex.normSq_eq_norm_sq,Real.log_pow,Real.log_mul hf.ne' hc.ne'] at hl
  have he : Real.log (((1+(Polynomial.X : ℂ[X]) • (complexGram X).map Polynomial.C).det).coeff q).re =
      ∑ i, positiveSpectrumLog ((complexGram_posSemidef X).1.eigenvalues₀ i) := by
    rw [← hr]
    exact gram_rank_coefficient_log_eq_ordered_positive_sum X
  rw [he] at hl
  norm_num only [Nat.cast_ofNat] at hl
  have hu : (∑ i, positiveSpectrumLog ((complexGram_posSemidef X).1.eigenvalues₀ i)) ≤
      ∑ i, (complexGram_posSemidef X).1.eigenvalues₀ i :=
    Finset.sum_le_sum (fun i _ => positiveSpectrumLog_le_self (complexGram_eigenvalues₀_nonneg X i))
  have ht : 0 ≤ ∑ i, (complexGram_posSemidef X).1.eigenvalues₀ i :=
    Finset.sum_nonneg (fun i _ => complexGram_eigenvalues₀_nonneg X i)
  apply abs_le.mpr
  constructor
  · linarith [neg_le_abs (Real.log ‖(X.submatrix f g).det‖),le_abs_self (Real.log (q.factorial : ℝ))]
  · linarith [abs_nonneg (Real.log (q.factorial : ℝ)),abs_nonneg (Real.log ‖(X.submatrix f g).det‖)]

end
