   (Matrix.of (fun i j => f i (z j))).det * (Matrix.of (fun i j => G i (z j))).det := by
   unfold determinantProduct G
   have hd : (Matrix.of (fun i j => w (z j) * g i (z j))).det =
       (∏ i, w (z i)) * (Matrix.of (fun i j => g i (z j))).det := by
