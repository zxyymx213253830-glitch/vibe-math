def coordinateMembershipCount (E : Set ℝ) (z : Fin n → ℝ) : ℕ :=
  by classical exact (Finset.univ.filter (fun i => z i ∈ E)).card

theorem split_andreief_count_generating (f g : Fin n → ℝ → ℝ)
    (hfg : ∀ i j,Integrable (fun y => f i y*g j y) μ)
    (E : Set ℝ) (hE : MeasurableSet E) (t : ℝ) :
    Matrix.det (fun i j => t*(∫ y in E,f i y*g j y ∂μ)+
      ∫ y in Eᶜ,f i y*g j y ∂μ)=
    (1/(n.factorial : ℝ))*∫ z : Fin n → ℝ,
      t^(coordinateMembershipCount E z)*
        (Matrix.of (fun i j => f i (z j))).det*(Matrix.of (fun i j => g i (z j))).det
          ∂(Measure.pi (fun _ : Fin n => μ)) := by
  classical
  let F : Fin n → ℝ → ℝ := fun i y => (if y ∈ E then t else 1)*f i y
  have hF : ∀ i j,Integrable (fun y => F i y*g j y) μ := by
    intro i j
    have he : (fun y => F i y*g j y)=E.piecewise
        (fun y => t*(f i y*g j y)) (fun y => f i y*g j y) := by
      funext y
      by_cases hy : y ∈ E <;> simp [F,Set.piecewise,hy,mul_assoc]
    rw [he]
    exact Integrable.piecewise hE ((hfg i j).const_mul t).integrableOn (hfg i j).integrableOn
  have hm : (fun i j => ∫ y,F i y*g j y ∂μ)=
      (fun i j => t*(∫ y in E,f i y*g j y ∂μ)+∫ y in Eᶜ,f i y*g j y ∂μ) := by
    funext i j
    have he : (fun y => F i y*g j y)=E.piecewise
        (fun y => t*(f i y*g j y)) (fun y => f i y*g j y) := by
      funext y
      by_cases hy : y ∈ E <;> simp [F,Set.piecewise,hy,mul_assoc]
    rw [he,integral_piecewise hE ((hfg i j).const_mul t).integrableOn (hfg i j).integrableOn,
      integral_const_mul]
  rw [← hm]
  change (Matrix.of (fun i j => ∫ y,F i y*g j y ∂μ)).det=_
  rw [Andreief.andreief F g hF]
  congr 1
  apply integral_congr_ae
  filter_upwards [] with z
  have hd : (Matrix.of (fun i j => F i (z j))).det=
      t^(coordinateMembershipCount E z)*(Matrix.of (fun i j => f i (z j))).det := by
    have he := Matrix.det_mul_row (fun j => if z j ∈ E then t else 1)
      (Matrix.of (fun i j => f i (z j)))
    simp only [Matrix.of_apply] at he
    change (Matrix.of (fun i j => (if z j ∈ E then t else 1)*f i (z j))).det=_
    rw [he]
    congr 1
    simp [coordinateMembershipCount,Finset.prod_ite]
  rw [hd]
end

theorem measurable_coordinateMembershipCount (E : Set ℝ) (hE : MeasurableSet E) :
    Measurable (coordinateMembershipCount (n := n) E) := by
  classical
  have he : coordinateMembershipCount (n := n) E=
      fun z => ∑ i : Fin n,if z i ∈ E then (1 : ℕ) else 0 := by
    funext z
    simp only [coordinateMembershipCount,Finset.card_eq_sum_ones,Finset.sum_filter]
  rw [he]
  exact Finset.measurable_sum _ (fun i _ => measurable_const.ite
    (hE.preimage (measurable_pi_apply i)) measurable_const)

theorem andreief_det_product_integrable (f g : Fin n → ℝ → ℝ)
    (hfg : ∀ i j,Integrable (fun y => f i y*g j y) μ) :
    Integrable (fun z : Fin n → ℝ =>
      (Matrix.of (fun i j => f i (z j))).det*(Matrix.of (fun i j => g i (z j))).det)
        (Measure.pi (fun _ : Fin n => μ)) := by
  classical
  have hi (σ τ : Equiv.Perm (Fin n)) : Integrable
      (fun z : Fin n → ℝ => (Equiv.Perm.sign σ : ℝ)*(Equiv.Perm.sign τ : ℝ)*
        ((∏ i,f (σ i) (z i))*(∏ i,g (τ i) (z i)))) (Measure.pi (fun _ : Fin n => μ)) := by
    have he : (fun z : Fin n → ℝ => (∏ i,f (σ i) (z i))*(∏ i,g (τ i) (z i)))=
        fun z => ∏ i,f (σ i) (z i)*g (τ i) (z i) := by
      funext z
      exact (Finset.prod_mul_distrib).symm
    apply Integrable.const_mul
    rw [he]
    exact Integrable.fintype_prod (fun i => hfg (σ i) (τ i))
  have hp (z : Fin n → ℝ) :
      (Matrix.of (fun i j => f i (z j))).det*(Matrix.of (fun i j => g i (z j))).det=
      ∑ τ : Equiv.Perm (Fin n),∑ σ : Equiv.Perm (Fin n),
        (Equiv.Perm.sign σ : ℝ)*(Equiv.Perm.sign τ : ℝ)*
          ((∏ i,f (σ i) (z i))*(∏ i,g (τ i) (z i))) := by
    rw [Matrix.det_apply',Matrix.det_apply',Finset.sum_mul_sum,Finset.sum_comm]
    apply Finset.sum_congr rfl
    intro τ _
    apply Finset.sum_congr rfl
    intro σ _
    simp only [Matrix.of_apply]
    ring
  simp_rw [hp]
  exact integrable_finsetSum _ (fun τ _ => integrable_finsetSum _ (fun σ _ => hi σ τ))

    (Matrix.of (fun i j => f i (z j))).det*(Matrix.of (fun i j => g i (z j))).det
      ∂(Measure.pi (fun _ : Fin n => μ))

  classical
  let ν := Measure.pi (fun _ : Fin n => μ)

    (Matrix.of (fun i j => f i (z j))).det*(Matrix.of (fun i j => g i (z j))).det

  apply Polynomial.funext
  intro t

      (Matrix.of (fun i j => ∫ y in Eᶜ,f i y*g j y ∂μ))).det).eval t=

  have hpart (z : Fin n → ℝ) : t^coordinateMembershipCount E z*D z=
