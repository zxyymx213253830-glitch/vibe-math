/-! Group permutation-invariant function weights by their injective image.
The target sum includes all cardinality-q subsets, including the empty case.
Codex finite algebra; no stochastic or analytic assumptions. -/
open Finset Function
namespace JinWishart
noncomputable section

def cardinalSubsetEnumeration {q m : ℕ} (S : {S : Finset (Fin m) // S.card = q}) :
    Fin q ≃ S.val := Fintype.equivOfCardEq (by simpa using S.property.symm)

def injectiveFunctionImage {q m : ℕ} (f : {f : Fin q → Fin m // Injective f}) :
    {S : Finset (Fin m) // S.card = q} :=
  ⟨univ.image f.val, by simp [card_image_of_injective _ f.property]⟩

theorem permutation_injective_image_fiber_equiv {q m : ℕ}
    (S : {S : Finset (Fin m) // S.card = q}) :
    Nonempty ((Equiv.Perm (Fin q)) ≃
      {f : {f : Fin q → Fin m // Injective f} // injectiveFunctionImage f = S}) := by
  classical
  let e := cardinalSubsetEnumeration S
  let F : Equiv.Perm (Fin q) → {f : {f : Fin q → Fin m // Injective f} // injectiveFunctionImage f = S} :=
    fun σ => ⟨⟨fun i => (e (σ i)).val, Subtype.val_injective.comp (e.injective.comp σ.injective)⟩, by
      apply Subtype.ext
      ext j
      simp only [injectiveFunctionImage,mem_image,mem_univ,true_and]
      constructor
      · rintro ⟨i,rfl⟩
        exact (e (σ i)).property
      · intro hj
        obtain ⟨i,hi⟩ := (e.surjective.comp σ.surjective) ⟨j,hj⟩
        exact ⟨i,congrArg Subtype.val hi⟩⟩
  have hF : Bijective F := by
    constructor
    · intro σ τ h
      apply Equiv.ext
      intro i
      apply e.injective
      apply Subtype.ext
      exact congrFun (congrArg (fun x => x.val.val) h) i
    · intro f
      have hm (i : Fin q) : f.val.val i ∈ S.val := by
        have h := congrArg Subtype.val f.property
        rw [← h]
        exact mem_image_of_mem _ (mem_univ i)
      let g : Fin q → S.val := fun i => ⟨f.val.val i,hm i⟩
      have hg : Injective g := by
        intro i j h
        exact f.val.property (congrArg Subtype.val h)
      let E : Fin q ≃ S.val := Equiv.ofBijective g ⟨hg,hg.surjective_of_finite e⟩
      refine ⟨E.trans e.symm, ?_⟩
      apply Subtype.ext
      apply Subtype.ext
      funext i
      change (e (e.symm (E i))).val = f.val.val i
      simp only [e.apply_symm_apply]
      rfl
  exact ⟨Equiv.ofBijective F hF⟩

theorem sum_functions_eq_factorial_sum_cardinal_subsets {q m : ℕ}
    (w : (Fin q → Fin m) → ℂ)
    (hz : ∀ f, ¬ Injective f → w f = 0)
    (hp : ∀ f (σ : Equiv.Perm (Fin q)), w (fun i => f (σ i)) = w f) :
    (∑ f : Fin q → Fin m, w f) = (q.factorial : ℂ) *
      ∑ S : {S : Finset (Fin m) // S.card = q},
        w (fun i => (cardinalSubsetEnumeration S i).val) := by
  classical
  have hr : (∑ f : Fin q → Fin m, w f) =
      ∑ f : {f : Fin q → Fin m // Injective f}, w f.val := by
    rw [← Finset.sum_subtype (univ.filter Injective) (by simp) w]
    exact (Finset.sum_filter_of_ne fun f _ hf => by
      by_contra h; exact hf (hz f h)).symm
  rw [hr,← Fintype.sum_fiberwise injectiveFunctionImage (fun f => w f.val)]
  rw [Finset.mul_sum]
  apply Finset.sum_congr rfl
  intro S hS
  let e := cardinalSubsetEnumeration S
  obtain ⟨E⟩ := permutation_injective_image_fiber_equiv S
  -- The fiber has q! elements; each is a permutation of the chosen enumeration.
  have hc : Fintype.card {f : {f : Fin q → Fin m // Injective f} // injectiveFunctionImage f = S} = q.factorial := by
    rw [← Fintype.card_congr E,Fintype.card_perm,Fintype.card_fin]
  have hw (f : {f : {f : Fin q → Fin m // Injective f} // injectiveFunctionImage f = S}) :
      w f.val.val = w (fun i => (e i).val) := by
    have hm (i : Fin q) : f.val.val i ∈ S.val := by
      have h := congrArg Subtype.val f.property
      rw [← h]
      exact mem_image_of_mem _ (mem_univ i)
    let g : Fin q → S.val := fun i => ⟨f.val.val i,hm i⟩
    have hg : Injective g := by
      intro i j h
      exact f.val.property (congrArg Subtype.val h)
    let E' : Fin q ≃ S.val := Equiv.ofBijective g ⟨hg,hg.surjective_of_finite e⟩
    have he : (fun i => (e ((E'.trans e.symm) i)).val) = f.val.val := by
      funext i
      change (e (e.symm (E' i))).val = f.val.val i
      simp only [e.apply_symm_apply]
      rfl
    rw [← he]
    exact hp (fun i => (e i).val) (E'.trans e.symm)
  simp_rw [hw]
  simp [hc,e]

end
end JinWishart

/-! Exact subset-grouped rectangular Cauchy--Binet over complex matrices.
One cardinality-q subset contributes once; no residual factorial.
Codex finite algebra, including q=0 and q>m. -/

theorem rectangular_minor_product_permutation_invariant {q m : ℕ}
    (A : Matrix (Fin q) (Fin m) ℂ) (B : Matrix (Fin m) (Fin q) ℂ)
    (f : Fin q → Fin m) (σ : Equiv.Perm (Fin q)) :
    (A.submatrix id (fun i => f (σ i))).det * (B.submatrix (fun i => f (σ i)) id).det =
      (A.submatrix id f).det * (B.submatrix f id).det := by
  have ha : A.submatrix id (fun i => f (σ i)) = (A.submatrix id f).submatrix id σ := rfl
  have hb : B.submatrix (fun i => f (σ i)) id = (B.submatrix f id).submatrix σ id := rfl
  rw [ha,hb,Matrix.det_permute',Matrix.det_permute]
  have hs : (((Equiv.Perm.sign σ : ℤ) : ℂ))*(((Equiv.Perm.sign σ : ℤ) : ℂ))=1 := by
    exact_mod_cast Int.isUnit_mul_self (Equiv.Perm.sign σ).isUnit
  calc
    _ = (((Equiv.Perm.sign σ : ℤ) : ℂ))*(((Equiv.Perm.sign σ : ℤ) : ℂ))*
        ((A.submatrix id f).det * (B.submatrix f id).det) := by ring
    _ = _ := by rw [hs,one_mul]

theorem det_mul_rectangular_sum_cardinal_subsets {q m : ℕ}
    (A : Matrix (Fin q) (Fin m) ℂ) (B : Matrix (Fin m) (Fin q) ℂ) :
    (A*B).det = ∑ S : {S : Finset (Fin m) // S.card = q},
      (A.submatrix id (fun i => (cardinalSubsetEnumeration S i).val)).det *
        (B.submatrix (fun i => (cardinalSubsetEnumeration S i).val) id).det := by
  classical
  have hg := sum_functions_eq_factorial_sum_cardinal_subsets
    (fun f => (A.submatrix id f).det * (B.submatrix f id).det)
    (fun f hf => by rw [det_submatrix_eq_zero_of_not_injective B f hf,mul_zero])
    (rectangular_minor_product_permutation_invariant A B)
  rw [← det_mul_rectangular_factorial_sum_functions] at hg
  exact mul_left_cancel₀ (by exact_mod_cast (Nat.factorial_ne_zero q)) hg

theorem gram_det_sum_squared_cardinal_minors {m q : ℕ}
    (B : Matrix (Fin m) (Fin q) ℂ) :
    (complexGram B).det = ∑ S : {S : Finset (Fin m) // S.card = q},
      star ((B.submatrix (fun i => (cardinalSubsetEnumeration S i).val) id).det) *
        (B.submatrix (fun i => (cardinalSubsetEnumeration S i).val) id).det := by
  classical
  have h := det_mul_rectangular_sum_cardinal_subsets B.conjTranspose B
  have he (S : {S : Finset (Fin m) // S.card = q}) :
      B.conjTranspose.submatrix id (fun i => (cardinalSubsetEnumeration S i).val) =
      (B.submatrix (fun i => (cardinalSubsetEnumeration S i).val) id).conjTranspose := rfl
  simpa only [he,Matrix.det_conjTranspose,complexGram] using h

end
end JinWishart

/-! The coefficient of det(I+t X^H X) is the full sum of squared q-minors.
Each row/column subset is included once. Codex finite algebra, all finite shapes. -/
open Matrix Polynomial Finset Function

theorem gram_principal_minor_eq_enumerated_gram {m n q : ℕ}
    (X : Matrix (Fin m) (Fin n) ℂ) (T : {T : Finset (Fin n) // T.card = q}) :
    ((complexGram X).submatrix (Subtype.val : T.val → Fin n) Subtype.val).det =
      (complexGram (X.submatrix id (fun j => (cardinalSubsetEnumeration T j).val))).det := by
  classical
  rw [← Matrix.det_submatrix_equiv_self (cardinalSubsetEnumeration T)]
  rfl

theorem gram_polynomial_coefficient_eq_sum_squared_minors {m n : ℕ}
    (X : Matrix (Fin m) (Fin n) ℂ) (q : ℕ) :
    (((1+(Polynomial.X : ℂ[X]) • (complexGram X).map Polynomial.C).det).coeff q) =
      ∑ T : {T : Finset (Fin n) // T.card = q},
        ∑ S : {S : Finset (Fin m) // S.card = q},
          star ((X.submatrix (fun i => (cardinalSubsetEnumeration S i).val)
            (fun j => (cardinalSubsetEnumeration T j).val)).det) *
          (X.submatrix (fun i => (cardinalSubsetEnumeration S i).val)
            (fun j => (cardinalSubsetEnumeration T j).val)).det := by
  classical
  rw [Matrix.coeff_det_one_add_X_smul_eq_sum_minors]
  rw [Finset.sum_subtype (p := fun T : Finset (Fin n) => T.card = q)
    (univ.powersetCard q) (by simp [Finset.mem_powersetCard])]
  apply Finset.sum_congr rfl
  intro T hT
  rw [gram_principal_minor_eq_enumerated_gram,gram_det_sum_squared_cardinal_minors]
  rfl

theorem gram_polynomial_coefficient_re_eq_sum_normSq_minors {m n : ℕ}
    (X : Matrix (Fin m) (Fin n) ℂ) (q : ℕ) :
    ((((1+(Polynomial.X : ℂ[X]) • (complexGram X).map Polynomial.C).det).coeff q)).re =
      ∑ T : {T : Finset (Fin n) // T.card = q},
        ∑ S : {S : Finset (Fin m) // S.card = q},
          Complex.normSq ((X.submatrix (fun i => (cardinalSubsetEnumeration S i).val)
            (fun j => (cardinalSubsetEnumeration T j).val)).det) := by
  classical
  have h := congrArg Complex.re (gram_polynomial_coefficient_eq_sum_squared_minors X q)
  simpa [Complex.mul_re,Complex.star_def,Complex.normSq_apply] using h

end
