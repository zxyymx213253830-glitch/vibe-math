theorem meanPositiveNoncentrality_sum_eq_energy {m n : ℕ}
 (M : Matrix (Fin m) (Fin n) ℂ) :
 (∑ i : Fin M.rank, meanPositiveNoncentrality M i) =
 ∑ a : Fin m × Fin n, ‖M a.1 a.2‖^2 := by
 let h := (complexGram_posSemidef M).1
 have heig : (∑ i : Fin (Fintype.card (Fin n)), h.eigenvalues₀ i) =
     (complexGram M).trace.re := by
   let e : Fin (Fintype.card (Fin n)) ≃ Fin n :=
     Fintype.equivOfCardEq (Fintype.card_fin (Fintype.card (Fin n)))
   have hs : (∑ i, h.eigenvalues₀ i) = ∑ i, h.eigenvalues i := by
     simpa only [Matrix.IsHermitian.eigenvalues,e,Equiv.symm_apply_apply] using
       (Equiv.sum_comp e (fun i => h.eigenvalues i))
   have ht := congrArg Complex.re h.trace_eq_sum_eigenvalues
   simp only [Complex.re_sum] at ht
   change (complexGram M).trace.re = ∑ i, h.eigenvalues i at ht
   exact hs.trans ht.symm
 have hr : M.rank ≤ n := Matrix.rank_le_width M
 have hn : M.rank + (n-M.rank) = n := Nat.add_sub_of_le hr
 let e : Fin (M.rank+(n-M.rank)) ≃ Fin (Fintype.card (Fin n)) :=
   finCongr (by simpa using hn)
 let f : Fin (M.rank+(n-M.rank)) → ℝ := fun i => h.eigenvalues₀ (e i)
 have hf : (∑ i : Fin (Fintype.card (Fin n)), h.eigenvalues₀ i) = ∑ i, f i := by
   simpa only [f] using (Equiv.sum_comp e (fun i => h.eigenvalues₀ i)).symm
 have hsplit := Fin.sum_univ_add f
 have hhead : (∑ i : Fin M.rank, f (Fin.castAdd (n-M.rank) i)) =
     ∑ i : Fin M.rank, meanPositiveNoncentrality M i := by
   apply Finset.sum_congr rfl
   intro i _
   unfold f e meanPositiveNoncentrality h
   congr 1
 have htail : (∑ i : Fin (n-M.rank), f (Fin.natAdd M.rank i)) = 0 := by
   apply Finset.sum_eq_zero
   intro i _
   apply (meanGram_eigenvalue_zero_iff_rank_le_index M _).2
   simp [f,e]
 rw [hf,hsplit,hhead,htail,add_zero] at heig
 rw [complexGram_trace_re] at heig
 rw [Fintype.sum_prod_type]
 exact heig
