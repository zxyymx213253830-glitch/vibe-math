theorem binomial_overlap_sum_fin (n : ℕ) (i j : Fin n) :
    (∑ k : Fin n,i.val.choose k.val*j.val.choose k.val)=(i.val+j.val).choose j.val := by
  classical
  rw [Fin.sum_univ_eq_sum_range (fun k : ℕ => i.val.choose k*j.val.choose k) n]
  have h := Nat.add_choose_eq i.val j.val j.val
  rw [Finset.Nat.sum_antidiagonal_eq_sum_range_succ_mk] at h
  have ht : (∑ k ∈ Finset.range (j.val+1),i.val.choose k*j.val.choose k)=
      ∑ k ∈ Finset.range n,i.val.choose k*j.val.choose k := by
    apply Finset.sum_subset (Finset.range_mono (by omega))
    intro k hk hnot
    have hjk : j.val<k := by simp only [Finset.mem_range] at hnot;omega
    rw [Nat.choose_eq_zero_of_lt hjk,mul_zero]
  rw [← ht, h]
  apply Finset.sum_congr rfl
  intro k hk
  rw [Nat.choose_symm (by simpa only [Finset.mem_range_succ_iff] using hk)]

def factorialHankel (n : ℕ) : Matrix (Fin n) (Fin n) ℂ :=
  fun i j => ((i.val+j.val).factorial : ℂ)

def factorialPascalFactor (n : ℕ) : Matrix (Fin n) (Fin n) ℂ :=
  fun i j => (i.val.factorial : ℂ)*(i.val.choose j.val : ℂ)

theorem factorialHankel_gram_factorization (n : ℕ) :
    factorialHankel n=factorialPascalFactor n*(factorialPascalFactor n).transpose := by
  classical
  ext i j
  rw [Matrix.mul_apply]
  change ((i.val+j.val).factorial : ℂ)=
    ∑ k : Fin n,(i.val.factorial : ℂ)*(i.val.choose k.val : ℂ)*
      ((j.val.factorial : ℂ)*(j.val.choose k.val : ℂ))
  have hs : (∑ k : Fin n,(i.val.choose k.val : ℂ)*(j.val.choose k.val : ℂ))=
      ((i.val+j.val).choose j.val : ℂ) := by
    exact_mod_cast binomial_overlap_sum_fin n i j
  calc
    _=((i.val+j.val).choose j.val : ℂ)*(i.val.factorial : ℂ)*(j.val.factorial : ℂ) := by
      exact_mod_cast (Nat.add_choose_mul_factorial_mul_factorial i.val j.val).symm
    _=(i.val.factorial : ℂ)*(j.val.factorial : ℂ)*
        (∑ k : Fin n,(i.val.choose k.val : ℂ)*(j.val.choose k.val : ℂ)) := by rw [hs];ring
    _=_ := by
      rw [Finset.mul_sum]
      apply Finset.sum_congr rfl
      intro k _
      ring

theorem factorialHankel_det (n : ℕ) :
    (factorialHankel n).det=∏ i : Fin n,(i.val.factorial : ℂ)^2 := by
  classical
  have htri : (factorialPascalFactor n).IsLowerTriangular := by
    intro i j hij
    have hij' : i.val<j.val := hij
    change (i.val.factorial : ℂ)*(i.val.choose j.val : ℂ)=0
    rw [Nat.choose_eq_zero_of_lt hij',Nat.cast_zero,mul_zero]
  have hdet : (factorialPascalFactor n).det=∏ i : Fin n,(i.val.factorial : ℂ) := by
    rw [Matrix.det_of_isLowerTriangular _ htri]
    simp [factorialPascalFactor]
  rw [factorialHankel_gram_factorization,Matrix.det_mul,Matrix.det_transpose,hdet]
  rw [← pow_two,Finset.prod_pow]

end

theorem monic_polynomial_moment_det (n : ℕ) (F : ℂ[X] →ₗ[ℂ] ℂ)
    (p : Fin n → ℂ[X]) (hd : ∀ i,(p i).natDegree=i.val) (hm : ∀ i,(p i).Monic) :
    Matrix.det (fun i j : Fin n => F (p i*p j))=
      Matrix.det (fun i j : Fin n => F (X^(i.val+j.val))) := by
  classical
  let T : Matrix (Fin n) (Fin n) ℂ := fun i j => (p i).coeff j.val
  let H : Matrix (Fin n) (Fin n) ℂ := fun i j => F (X^(i.val+j.val))
  have hp (i : Fin n) : p i=∑ k : Fin n,(p i).coeff k.val • (X : ℂ[X])^k.val := by
    rw [Fin.sum_univ_eq_sum_range (fun k : ℕ => (p i).coeff k • (X : ℂ[X])^k) n]
    simpa only [smul_eq_C_mul] using
      (p i).as_sum_range_C_mul_X_pow' (by rw [hd];exact i.isLt)
  have hdet : T.det=1 := by
    rw [← Matrix.det_transpose]
    exact Matrix.det_matrixOfPolynomials p hd hm
  have heq : (fun i j : Fin n => F (p i*p j))=T*H*T.transpose := by
    ext i j
    rw [Matrix.mul_assoc,Matrix.mul_apply]
    change F (p i*p j)=∑ k : Fin n,(p i).coeff k.val*
      (∑ l : Fin n,F (X^(k.val+l.val))*(p j).coeff l.val)
    conv_lhs => rw [hp i,hp j,Finset.sum_mul]
    simp only [Finset.mul_sum,map_sum]
    apply Finset.sum_congr rfl
    intro k _
    apply Finset.sum_congr rfl
    intro l _
    simp only [smul_mul_assoc,mul_smul_comm,smul_smul,map_smul,RingHom.id_apply,
      ← pow_add,smul_eq_mul]
    ring
  rw [heq,Matrix.det_mul,Matrix.det_mul,Matrix.det_transpose,hdet,one_mul,mul_one]

def factorialMomentFunctional : ℂ[X] →ₗ[ℂ] ℂ :=
  Polynomial.lsum (fun k => (LinearMap.id : ℂ →ₗ[ℂ] ℂ).smulRight (k.factorial : ℂ))

@[simp] private theorem factorialMoment_monomial (k : ℕ) (a : ℂ) :
    factorialMomentFunctional (monomial k a)=a*(k.factorial : ℂ) := by
  simp [factorialMomentFunctional,Polynomial.lsum_apply,Polynomial.sum_monomial_index]

@[simp] private theorem factorialMoment_C (a : ℂ) : factorialMomentFunctional (C a)=a := by
  simpa using factorialMoment_monomial 0 a

@[simp] private theorem factorialMoment_one : factorialMomentFunctional 1=1 := by
  simpa using factorialMoment_C 1

private theorem factorialMoment_eval_derivative (p : ℂ[X]) :
    factorialMomentFunctional p=p.eval 0+factorialMomentFunctional p.derivative := by
  induction p using Polynomial.induction_on' with
  | add p q hp hq =>
    simp only [map_add,eval_add,derivative_add,hp,hq]
    ring
  | monomial n a =>
    cases n with
    | zero => simp
    | succ n =>
      simp [derivative_monomial,Nat.factorial_succ]
      ring

private theorem factorialMoment_shift_succ (k : ℕ) (x : ℂ) :
    factorialMomentFunctional ((X+C x)^(k+1))=
      x^(k+1)+(k+1 : ℂ)*factorialMomentFunctional ((X+C x)^k) := by
  have h := factorialMoment_eval_derivative ((X+C x)^(k+1))
  simpa only [eval_pow,eval_add,eval_X,eval_C,zero_add,derivative_X_add_C_pow,
    Nat.add_sub_cancel,← smul_eq_C_mul,map_smul,RingHom.id_apply,smul_eq_mul,Nat.cast_add,Nat.cast_one] using h

theorem factorialMoment_X_pow (k : ℕ) :
    factorialMomentFunctional ((X : ℂ[X])^k)=(k.factorial : ℂ) := by
  rw [← monomial_one_right_eq_X_pow,factorialMoment_monomial,one_mul]

theorem upperGammaNat_eq_factorialMoment (k : ℕ) (x : ℝ) (hx : 0≤x) :
    upperGammaNat k x=(Real.exp (-x) : ℂ)*
      factorialMomentFunctional ((X+C (x : ℂ))^k) := by
  induction k with
  | zero => simp [upperGammaNat_zero,← monomial_zero_left]
  | succ k ih =>
    rw [upperGammaNat_succ k hx,ih,factorialMoment_shift_succ]
    have hp : (x : ℂ)^((k+1 : ℕ) : ℂ)=(x : ℂ)^(k+1) := Complex.cpow_natCast _ _
    rw [hp]
    push_cast
    ring

end
end JinWishart
