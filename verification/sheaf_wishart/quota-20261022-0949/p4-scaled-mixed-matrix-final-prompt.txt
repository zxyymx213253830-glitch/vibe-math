theorem weightedMixedMomentMatrix_det (p L q : ℕ) (lambda : Fin L → ℝ)
 (hl : ∀ j, 0 ≤ lambda j) (c : ℝ) (hc : 0 < c) :

 (∏ i : Fin (p+L), (c^(q+i.val+1))⁻¹) *
 (∏ j : Fin p ⊕ Fin L, weightedMixedColumnFactor p L lambda c j) *
 (mixedNoncentralMomentMatrix q p L (fun b => lambda b/c)).det := by
 let e := finSumFinEquiv (m := p) (n := L)
 let A := (mixedNoncentralMomentMatrix q p L (fun b => lambda b/c)).submatrix e.symm e.symm
 let r : Fin (p+L) → ℝ := fun i => (c^(q+i.val+1))⁻¹
 let w : Fin (p+L) → ℝ := fun j => weightedMixedColumnFactor p L lambda c (e.symm j)
 have hm : weightedMixedMomentMatrix p L q lambda c =
     Matrix.of (fun i j => r i * (w j * A i j)) := by
   ext i j
   simp only [Matrix.of_apply,r,w,A,Matrix.submatrix_apply,e]
   rw [weightedMixedMomentMatrix_entry p L q lambda hl c hc]
   ring
 have hrow : (Matrix.of (fun i j => r i * (w j * A i j))).det =
     (∏ i, r i) * (Matrix.of (fun i j => w j * A i j)).det := by
   simpa only [Matrix.of_apply] using
     Matrix.det_mul_column r (Matrix.of (fun i j => w j * A i j))
 have hcol : (Matrix.of (fun i j => w j * A i j)).det = (∏ j, w j) * A.det := by
   exact Matrix.det_mul_row w A
 have hw : (∏ j, w j) = ∏ j : Fin p ⊕ Fin L, weightedMixedColumnFactor p L lambda c j := by
   exact e.symm.prod_comp (weightedMixedColumnFactor p L lambda c)

 rw [Matrix.det_submatrix_equiv_self]
