def factorialLaguerre (n : ℕ) : ℂ[X] :=
  ∑ k ∈ Finset.range (n+1), C (((-1 : ℂ)^(n-k))*(n.factorial : ℂ)*
    (n.choose k : ℂ)/(k.factorial : ℂ))*X^k

private theorem alternatingPolynomialSum (n : ℕ) (p : ℂ[X]) (hp : p.natDegree<n) :
    (∑ k ∈ Finset.range (n+1),(-1 : ℂ)^(n-k)*(n.choose k : ℂ)*p.eval ((k : ℂ)+1))=0 := by
  have h := congrFun (Polynomial.fwdDiff_iter_eq_zero_of_degree_lt hp) (1 : ℂ)
  rw [fwdDiff_iter_eq_sum_shift] at h
  simpa [zsmul_eq_mul,Int.cast_mul,Int.cast_pow,Int.cast_neg,Int.cast_one,
    Nat.cast_smul_eq_nsmul,add_comm] using h

theorem factorial_alternating_moment_sum (n j : ℕ) (hj : j<n) :
    (∑ k ∈ Finset.range (n+1),(-1 : ℂ)^(n-k)*(n.choose k : ℂ)*
      ((j+k).factorial : ℂ)/(k.factorial : ℂ))=0 := by
  have h := alternatingPolynomialSum n (ascPochhammer ℂ j) (by simpa using hj)
  have hr (k : ℕ) : ((j+k).factorial : ℂ)/(k.factorial : ℂ)=
      (ascPochhammer ℂ j).eval ((k : ℂ)+1) := by
    apply (div_eq_iff (by exact_mod_cast Nat.factorial_ne_zero k : (k.factorial : ℂ)≠0)).2
    simpa [Nat.add_comm,mul_comm] using (factorial_mul_ascPochhammer ℂ k j).symm
  calc
    _ = ∑ k ∈ Finset.range (n+1),(-1 : ℂ)^(n-k)*(n.choose k : ℂ)*
        (ascPochhammer ℂ j).eval ((k : ℂ)+1) := by
      apply Finset.sum_congr rfl
      intro k _
      rw [mul_div_assoc,hr]
    _ = 0 := h

private theorem factorialLaguerre_degree_le (n : ℕ) : (factorialLaguerre n).natDegree≤n := by
  unfold factorialLaguerre
  apply natDegree_sum_le_of_forall_le
  intro k hk
  exact (natDegree_C_mul_X_pow_le _ _).trans (by simpa using (Nat.le_of_lt_succ (Finset.mem_range.mp hk)))

private theorem factorialLaguerre_coeff_top (n : ℕ) : (factorialLaguerre n).coeff n=1 := by
  unfold factorialLaguerre
  simp [coeff_sum,coeff_C_mul_X_pow,Finset.sum_eq_single n,Nat.factorial_ne_zero]

theorem factorialLaguerre_monic (n : ℕ) : (factorialLaguerre n).Monic := by
  exact monic_of_natDegree_le_of_coeff_eq_one n (factorialLaguerre_degree_le n)
    (factorialLaguerre_coeff_top n)

theorem factorialLaguerre_natDegree (n : ℕ) : (factorialLaguerre n).natDegree=n := by
  apply le_antisymm (factorialLaguerre_degree_le n)
  exact le_natDegree_of_ne_zero (by rw [factorialLaguerre_coeff_top];norm_num)

theorem factorialLaguerre_eval_neg (n : ℕ) (x : ℂ) :
    (-1 : ℂ)^n*(factorialLaguerre n).eval (-x)=
      (n.factorial : ℂ)*∑ k ∈ Finset.range (n+1),(n.choose k : ℂ)*x^k/(k.factorial : ℂ) := by

theorem monic_functional_columns_det (n : ℕ) (G : Fin n → ℂ[X] →ₗ[ℂ] ℂ)
    (p : Fin n → ℂ[X]) (hd : ∀ i,(p i).natDegree=i.val) (hm : ∀ i,(p i).Monic) :
    Matrix.det (fun i j : Fin n => G i (p j)) =
      Matrix.det (fun i j : Fin n => G i (X^j.val)) := by
  classical
  let T : Matrix (Fin n) (Fin n) ℂ := fun i j => (p i).coeff j.val
  let B : Matrix (Fin n) (Fin n) ℂ := fun i j => G i (X^j.val)
  have hp (j : Fin n) : p j=∑ k : Fin n,(p j).coeff k.val • (X : ℂ[X])^k.val := by
    rw [Fin.sum_univ_eq_sum_range (fun k : ℕ => (p j).coeff k • (X : ℂ[X])^k) n]
    simpa only [smul_eq_C_mul] using
      (p j).as_sum_range_C_mul_X_pow' (by rw [hd];exact j.isLt)
  have hT : T.det=1 := by
    rw [← Matrix.det_transpose]
    exact Matrix.det_matrixOfPolynomials p hd hm
  have he : Matrix.of (fun i j : Fin n => G i (p j))=B*T.transpose := by
    ext i j
    rw [Matrix.mul_apply]
    change G i (p j)=∑ k : Fin n,G i (X^k.val)*(p j).coeff k.val
    conv_lhs => rw [hp j,map_sum]
    apply Finset.sum_congr rfl
    intro k _
    simp [map_smul,mul_comm]
  change (Matrix.of (fun i j : Fin n => G i (p j))).det=B.det
  rw [he,Matrix.det_mul,Matrix.det_transpose,hT,mul_one]

theorem christoffel_moment_det (n : ℕ) (F : ℂ[X] →ₗ[ℂ] ℂ) (p : ℂ[X])
    (hd : p.natDegree=n) (hm : p.Monic)
    (ho : ∀ i : Fin n,F (X^i.val*p)=0) (z : ℂ) :
    Matrix.det (fun i j : Fin n => F ((X-C z)*X^(i.val+j.val))) =
      (-1 : ℂ)^n*p.eval z*Matrix.det (fun i j : Fin n => F (X^(i.val+j.val))) := by
  classical
  let G : Fin (n+1) → ℂ[X] →ₗ[ℂ] ℂ := Fin.cases (Polynomial.leval z)
    (fun i => F.comp (LinearMap.mulLeft ℂ ((X : ℂ[X])^i.val)))
  let Q : Fin (n+1) → ℂ[X] := Fin.cases 1 (fun j => X^j.val*(X-C z))
  let P : Fin (n+1) → ℂ[X] := Fin.lastCases p (fun j => X^j.val)
  have hdQ (j : Fin (n+1)) : (Q j).natDegree=j.val := by
    refine Fin.cases ?_ (fun j => ?_) j
    · simp [Q]
    · simp [Q,Polynomial.natDegree_mul, (monic_X_sub_C z).ne_zero]
  have hmQ (j : Fin (n+1)) : (Q j).Monic := by
    refine Fin.cases ?_ (fun j => ?_) j
    · exact monic_one
    · exact (monic_X_pow _).mul (monic_X_sub_C z)
  have hdP (j : Fin (n+1)) : (P j).natDegree=j.val := by
    refine Fin.lastCases ?_ (fun j => ?_) j
    · simpa [P] using hd
    · simp [P]
  have hmP (j : Fin (n+1)) : (P j).Monic := by
    refine Fin.lastCases ?_ (fun j => ?_) j
    · simpa [P] using hm
    · simpa [P] using monic_X_pow (R := ℂ) j.val
  have hQ : Matrix.det (fun i j : Fin (n+1) => G i (Q j))=
      Matrix.det (fun i j : Fin n => F ((X-C z)*X^(i.val+j.val))) := by
    have hminor : (Matrix.of (fun i j : Fin (n+1) => G i (Q j))).submatrix Fin.succ Fin.succ=
        Matrix.of (fun i j : Fin n => F ((X-C z)*X^(i.val+j.val))) := by
      ext i j
      simp [G,Q,Matrix.submatrix_apply,pow_add,mul_comm,mul_left_comm,mul_assoc]
    change (Matrix.of (fun i j : Fin (n+1) => G i (Q j))).det= _
    rw [Matrix.det_succ_row_zero (Matrix.of (fun i j : Fin (n+1) => G i (Q j))),Fin.sum_univ_succ]
    simp only [Fin.succAbove_zero]
    rw [hminor]
    simp [G,Q,Polynomial.leval_apply]
    rfl
  have hP : Matrix.det (fun i j : Fin (n+1) => G i (P j))=
      (-1 : ℂ)^n*p.eval z*Matrix.det (fun i j : Fin n => F (X^(i.val+j.val))) := by
    have hminor : (Matrix.of (fun i j : Fin (n+1) => G i (P j))).submatrix Fin.succ Fin.castSucc=
        Matrix.of (fun i j : Fin n => F (X^(i.val+j.val))) := by
      ext i j
      simp [G,P,Matrix.submatrix_apply,← pow_add]
    change (Matrix.of (fun i j : Fin (n+1) => G i (P j))).det= _
    rw [Matrix.det_succ_column (Matrix.of (fun i j : Fin (n+1) => G i (P j))) (Fin.last n),Fin.sum_univ_succ]
    simp only [Fin.succAbove_zero,Fin.succAbove_last]
    rw [hminor]
    simp [G,P,Polynomial.leval_apply,ho]
    exact Or.inl rfl
  exact hQ.symm.trans ((monic_functional_columns_det (n+1) G Q hdQ hmQ).trans
    ((monic_functional_columns_det (n+1) G P hdP hmP).symm.trans hP))

end
